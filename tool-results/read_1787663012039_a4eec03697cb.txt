   279→  function importJobsCsv(file: File) {
   280→    const reader = new FileReader();
   281→    reader.onload = async () => {
   282→      const text = String(reader.result ?? "");
   283→      const lines = text.split("\n").filter((l) => l.trim());
   284→      if (lines.length < 2) {
   285→        toast.error("CSV must have a header and at least one row");
   286→        return;
   287→      }
   288→      const headers = lines[0].split(",").map((h) => h.trim().toLowerCase().replace(/"/g, ""));
   289→      let imported = 0;
   290→      let failed = 0;
   291→      let customers = customersData?.customers ?? [];
   292→      if (customers.length === 0) {
   293→        try {
   294→          const resp = await customersApi.list();
   295→          customers = resp.customers;
   296→        } catch { /* ignore */ }
   297→      }
   298→      const technicians = usersData?.users ?? [];
   299→      const techMap = new Map(technicians.map((t) => [t.fullName.toLowerCase(), t.id]));
   300→
   301→      for (const line of lines.slice(1)) {
   302→        const cells = line.split(",").map((c) => c.trim().replace(/^"|"$/g, "").replace(/""/g, '"'));
   303→        const get = (key: string) => {
   304→          const idx = headers.indexOf(key);
   305→          return idx >= 0 ? cells[idx] : "";
   306→        };
   307→        const title = get("title");
   308→        const customerName = get("customer name") || get("customer");
   309→        if (!title || !customerName) { failed++; continue; }
   310→        const customer = customers.find((c) => c.name.toLowerCase() === customerName.toLowerCase());
   311→        if (!customer) { failed++; continue; }
   312→        const assignedTo = get("assigned to");
   313→        const assignedToId = assignedTo ? (techMap.get(assignedTo.toLowerCase()) || "") : "";
   314→        const statusVal = (get("status") as JobStatus) || "Pending";
   315→        try {
   316→          // The create endpoint intentionally omits `status` from its
   317→          // payload type for security; we set it via a separate update
   318→          // after the job is created, when the user picked a non-default
   319→          // status in the import template.
   320→          const created = await jobsApi.create({
   321→            title,
   322→            customerId: customer.id,
   323→            assignedToId: assignedToId || undefined,
   324→            priority: (get("priority") as Priority) || "Normal",
   325→            description: get("description") || undefined,
   326→            deliveryDate: get("delivery date") ? new Date(get("delivery date")).toISOString() : undefined,
   327→          });
   328→          if (statusVal && statusVal !== "Pending") {
   329→            try {
   330→              await jobsApi.update(created.job.id, { status: statusVal });
   331→            } catch {
   332→              /* best-effort status patch */
   333→            }
   334→          }
   335→          imported++;
   336→        } catch (e) {
   337→          failed++;
   338→          console.error("Import error:", e);
   339→        }
   340→      }
   341→      toast.success(`Imported ${imported} job(s)${failed > 0 ? `, ${failed} failed` : ""}`);
   342→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   343→    };
   344→    reader.readAsText(file);
   345→  }

   347→  async function duplicateJob(jobId: string) {
   348→    try {
   349→      const original = jobs.find((j) => j.id === jobId);
   350→      if (!original) {
   351→        toast.error("Job not found");
   352→        return;
   353→      }
   354→      const created = await jobsApi.create({
   355→        title: `${original.title} (copy)`,
   356→        customerId: original.customerId,
   357→        priority: original.priority,
   358→        description: original.description ?? undefined,
   359→        deliveryDate: original.deliveryDate ?? undefined,
   360→      });
   361→      toast.success(`Duplicated as ${created.job.orderNumber}`, {
   362→        description: created.job.title,
   363→      });
   364→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   365→      queryClient.invalidateQueries({ queryKey: ["stats"] });
   366→    } catch (e) {
   367→      toast.error(e instanceof Error ? e.message : "Failed to duplicate job");
   368→    }
   369→  }

  1490→/* ============ Calendar View — simple month grid ============ */
