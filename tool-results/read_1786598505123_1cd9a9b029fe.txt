   189→   189→function fmtDate(s?: string | null): string {
   190→   190→  if (!s) return "—";
   191→   191→  try {
   192→   192→    return new Date(s).toLocaleDateString(undefined, {
   193→   193→      year: "numeric",
   194→   194→      month: "short",
   195→   195→      day: "numeric",
   196→   196→    });
   197→   197→  } catch {
   198→   198→    return "—";
   199→   199→  }
   200→   200→}
   201→   201→
   202→   202→function fmtDateTime(s?: string | null): string {
   203→   203→  if (!s) return "—";
   204→   204→  try {
   205→   205→    return new Date(s).toLocaleString(undefined, {
   206→   206→      month: "short",
   207→   207→      day: "numeric",
   208→   208→      hour: "2-digit",
   209→   209→      minute: "2-digit",
   210→   210→    });
   211→   211→  } catch {
   212→   212→    return "—";
   213→   213→  }
   214→   214→}
   215→   215→
   216→   216→function toInputDate(s?: string | null): string {
   217→   217→  if (!s) return "";
   218→   218→  try {
   219→   219→    const d = new Date(s);
   220→   220→    if (Number.isNaN(d.getTime())) return "";
   221→   221→    return d.toISOString().slice(0, 10);
   222→   222→  } catch {
   223→   223→    return "";
   224→   224→  }
   225→   225→}
   226→   226→
   227→   227→function toInputDateTimeLocal(s?: string | null): string {
   228→   228→  if (!s) return "";
   229→   229→  try {
   230→   230→    const d = new Date(s);
   231→   231→    if (Number.isNaN(d.getTime())) return "";
   232→   232→    // local time
   233→   233→    const offsetMs = d.getTimezoneOffset() * 60000;
   234→   234→    return new Date(d.getTime() - offsetMs).toISOString().slice(0, 16);
   235→   235→  } catch {
   236→   236→    return "";
   237→   237→  }
   238→   238→}
