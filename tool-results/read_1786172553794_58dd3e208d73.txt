    34→export function ElevationCanvas({ data, height = 500, onUpdateModules }: ElevationCanvasProps) {
    35→  const { modules, overallDimensions } = data;
    36→  const totalW = overallDimensions.width;
    37→  const totalH = overallDimensions.height;
    38→
    39→  const [zoom, setZoom] = React.useState(1);
    40→  const [selectedId, setSelectedId] = React.useState<string | null>(null);
    41→  const [lockedIds, setLockedIds] = React.useState<Set<string>>(new Set());
    42→  const [contextMenu, setContextMenu] = React.useState<{ id: string; x: number; y: number } | null>(null);
    43→
    44→  const dragData = React.useRef<{
    45→    id: string; type: "move" | "resize"; edge?: string;
    46→    startPxX: number; startPxY: number;
    47→    origX: number; origY: number; origW: number; origH: number;
    48→  } | null>(null);
    49→  const innerRef = React.useRef<HTMLDivElement>(null);
    50→
    51→  const basePxPerMm = Math.min(700 / Math.max(totalW, 1), 350 / Math.max(totalH, 1));
    52→  const pxPerMm = basePxPerMm * zoom;
    53→  const wallPxW = totalW * pxPerMm;
    54→  const wallPxH = totalH * pxPerMm;
    55→  const PAD = 40;
    56→  const innerW = Math.max(wallPxW + PAD * 2, 700);
    57→  const innerH = Math.max(wallPxH + PAD * 2, 400);
    58→  const mm = (v: number) => v * pxPerMm;
    59→  const SNAP_THRESHOLD_MM = 50; // snap when within 50mm of another module's edge
    60→
    61→  // --- Snap helper: find nearby edges to snap to ---
    62→  function getSnappedPos(modId: string, newX: number, newY: number, newW: number, newH: number): { x: number; y: number } {
    63→    let snapX = newX, snapY = newY;
    64→    const myEdges = {
    65→      left: newX, right: newX + newW,
    66→      top: newY, bottom: newY + newH,
    67→      centerX: newX + newW / 2, centerY: newY + newH / 2,
    68→    };
    69→    for (const other of modules) {
    70→      if (other.id === modId) continue;
    71→      const oEdges = {
    72→        left: other.x, right: other.x + other.width,
    73→        top: other.y, bottom: other.y + other.height,
    74→        centerX: other.x + other.width / 2, centerY: other.y + other.height / 2,
    75→      };
    76→      // Check X snapping (left-to-left, right-to-right, left-to-right, right-to-left, center-to-center)
    77→      for (const [myKey, myVal] of Object.entries(myEdges)) {
    78→        if (!myKey.includes("x") && myKey !== "left" && myKey !== "right" && myKey !== "centerX") continue;
    79→        for (const [oKey, oVal] of Object.entries(oEdges)) {
    80→          if (!oKey.includes("x") && oKey !== "left" && oKey !== "right" && oKey !== "centerX") continue;
    81→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
    82→            if (myKey === "left" || myKey === "right" || myKey === "centerX") {
    83→              const diff = oVal - myVal;
    84→              snapX = newX + diff;
    85→            }
    86→          }
    87→        }
    88→      }
    89→      // Check Y snapping
    90→      for (const [myKey, myVal] of Object.entries(myEdges)) {
    91→        if (myKey !== "top" && myKey !== "bottom" && myKey !== "centerY") continue;
    92→        for (const [oKey, oVal] of Object.entries(oEdges)) {
    93→          if (oKey !== "top" && oKey !== "bottom" && oKey !== "centerY") continue;
    94→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
    95→            if (myKey === "top" || myKey === "bottom" || myKey === "centerY") {
    96→              const diff = oVal - myVal;
    97→              snapY = newY + diff;
    98→            }
    99→          }
   100→        }
   101→      }
   102→    }
   103→    // Also snap to wall edges
   104→    if (Math.abs(snapX) < SNAP_THRESHOLD_MM) snapX = 0;
   105→    if (Math.abs(snapY) < SNAP_THRESHOLD_MM) snapY = 0;
   106→    if (Math.abs(snapX + newW - totalW) < SNAP_THRESHOLD_MM) snapX = totalW - newW;
   107→    if (Math.abs(snapY + newH - totalH) < SNAP_THRESHOLD_MM) snapY = totalH - newH;
   108→    return { x: snapX, y: snapY };
   109→  }
   110→
   111→  function handleDrop(e: React.DragEvent) {
   112→    e.preventDefault();
   113→    const type = e.dataTransfer.getData("text/cabinet-type");
   114→    if (!type || !onUpdateModules) return;
   115→    const item = PALETTE.find((p) => p.type === type);
   116→    if (!item) return;
   117→    const inner = innerRef.current;
   118→    if (!inner) return;
   119→    const rect = inner.getBoundingClientRect();
   120→    const mmX = (e.clientX - rect.left - PAD) / pxPerMm;
   121→    const mmY = (e.clientY - rect.top - PAD) / pxPerMm;
   122→    const newCode = String(modules.length + 1).padStart(2, "0");
   123→    const newMod = createEmptyModule(newCode);
   124→    newMod.type = type as any;
   125→    newMod.width = item.w;
   126→    newMod.height = item.h;
   127→    newMod.x = Math.round(Math.max(0, Math.min(totalW - item.w, mmX - item.w / 2)));
   128→    newMod.y = Math.round(Math.max(0, Math.min(totalH - item.h, mmY - item.h / 2)));
   129→    onUpdateModules([...modules, newMod]);
   130→    setSelectedId(newMod.id);
   131→  }
   132→
   133→  function onModuleDown(e: React.PointerEvent, mod: CabinetModule) {
   134→    if (!onUpdateModules) return;
   135→    e.stopPropagation();
   136→    setSelectedId(mod.id);
   137→    if (lockedIds.has(mod.id)) return; // can't drag locked modules
   138→    dragData.current = {
   139→      id: mod.id, type: "move",
   140→      startPxX: e.clientX, startPxY: e.clientY,
   141→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   142→    };
   143→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   144→  }

   146→  function onResizeDown(e: React.PointerEvent, mod: CabinetModule, edge: string) {
   147→    e.stopPropagation();
   148→    setSelectedId(mod.id);
   149→    if (lockedIds.has(mod.id)) return;
   150→    dragData.current = {
   151→      id: mod.id, type: "resize", edge,
   152→      startPxX: e.clientX, startPxY: e.clientY,
   153→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   154→    };
   155→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   156→  }

   158→  function onCanvasMove(e: React.PointerEvent) {
   159→    const dd = dragData.current;
   160→    if (!dd || !onUpdateModules) return;
   161→    const dxMm = (e.clientX - dd.startPxX) / pxPerMm;
   162→    const dyMm = (e.clientY - dd.startPxY) / pxPerMm;
   163→    if (dd.type === "move") {
   164→      let newX = Math.max(0, Math.min(totalW - 50, Math.round(dd.origX + dxMm)));
   165→      let newY = Math.max(0, Math.min(totalH - 50, Math.round(dd.origY + dyMm)));
   166→      // Snap
   167→      const snapped = getSnappedPos(dd.id, newX, newY, dd.origW, dd.origH);
   168→      newX = snapped.x; newY = snapped.y;
   169→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, x: newX, y: newY } : m));
   170→    } else if (dd.type === "resize") {
   171→      const min = 100;
   172→      let nw = dd.origW, nh = dd.origH, nx = dd.origX, ny = dd.origY;
   173→      if (dd.edge === "right") nw = Math.max(min, Math.round(dd.origW + dxMm));
   174→      if (dd.edge === "left") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); }
   175→      if (dd.edge === "bottom") nh = Math.max(min, Math.round(dd.origH + dyMm));
   176→      if (dd.edge === "top") { nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   177→      // Corner resize (both dimensions)
   178→      if (dd.edge === "br") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   179→      if (dd.edge === "bl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   180→      if (dd.edge === "tr") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   181→      if (dd.edge === "tl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   182→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, width: nw, height: nh, x: nx, y: ny } : m));
   183→    }
   184→  }

   186→  function onCanvasUp() { dragData.current = null; }

   188→  function deleteModule(id: string) {
   189→    if (!onUpdateModules) return;
   190→    onUpdateModules(modules.filter((m) => m.id !== id).map((m, i) => ({ ...m, code: String(i + 1).padStart(2, "0") })));
   191→    setSelectedId(null);
   192→    setLockedIds((s) => { const n = new Set(s); n.delete(id); return n; });
   193→  }

   195→  function duplicateModule(id: string) {
   196→    if (!onUpdateModules) return;
   197→    const mod = modules.find((m) => m.id === id);
   198→    if (!mod) return;
   199→    const newCode = String(modules.length + 1).padStart(2, "0");
   200→    const copy: CabinetModule = {
   201→      ...mod,
   202→      id: crypto.randomUUID(),
   203→      code: newCode,
   204→      x: mod.x + 50,
   205→      y: mod.y + 50,
   206→      cuttingList: mod.cuttingList.map(c => ({ ...c, id: crypto.randomUUID() })),
   207→    };
   208→    onUpdateModules([...modules, copy]);
   209→    setSelectedId(copy.id);
   210→    toast.success(`Duplicated Module ${mod.code} → ${newCode}`);
   211→  }

   271→  function renderResizeNodes(mod: CabinetModule, w: number, h: number) {

   278→      { id: "tr", x: w - s/2, y: -s/2, cursor: "nesw-resize" },
   279→      { id: "r", x: w - s/2, y: h/2 - s/2, cursor: "ew-resize" },
   280→      { id: "br", x: w - s/2, y: h - s/2, cursor: "nwse-resize" },
   281→      { id: "b", x: w/2 - s/2, y: h - s/2, cursor: "ns-resize" },
   282→      { id: "bl", x: -s/2, y: h - s/2, cursor: "nesw-resize" },

   288→        style={{ left: n.x, top: n.y, width: s, height: s, cursor: n.cursor }}
   289→        onPointerDown={(e) => onResizeDown(e, mod, n.id)}
   290→      />
   291→    ));
   292→  }

   324→        onDrop={handleDrop}
   325→        onPointerMove={onCanvasMove}
   326→        onPointerUp={onCanvasUp}
   327→      >
   328→        <div ref={innerRef} className="relative shrink-0"
   329→          style={{ width: innerW, height: innerH,
   330→            backgroundImage: "linear-gradient(#f1f5f9 1px, transparent 1px), linear-gradient(90deg, #f1f5f9 1px, transparent 1px)",
   331→            backgroundSize: "20px 20px",
   332→          }}

   334→          {/* Wall outline */}
   335→          <div className="absolute border-2 border-dashed border-slate-300"

   342→            </div>

   344→
   345→          {/* Modules */}

   370→                {/* 8 resize nodes (CorelDRAW-style) */}
   371→                {renderResizeNodes(mod, w, h)}
   372→

   374→                {renderShape(mod, w, h)}
   375→              </div>
   376→            );
   377→          })}
   378→
   379→          {modules.length === 0 && (
   380→            <div className="absolute inset-0 flex items-center justify-center">
   381→              <span className="text-sm text-slate-400">Drag cabinet elements here →</span>
   382→            </div>
   383→          )}
   384→        </div>
   385→      </div>
   386→
   387→      {/* Context menu (right-click) */}
   388→      {contextMenu && (
   389→        <div
   390→          className="fixed z-50 min-w-[160px] rounded-lg border border-border bg-card py-1 shadow-xl"
   391→          style={{ left: contextMenu.x, top: contextMenu.y }}
   392→          onClick={(e) => e.stopPropagation()}
   393→          onPointerDown={(e) => e.stopPropagation()}

   395→          <button className="flex w-full items-center gap-2 px-3 py-1.5 text-xs hover:bg-muted"
   396→            onClick={() => { duplicateModule(contextMenu.id); setContextMenu(null); }}>
   397→            <Copy className="h-3.5 w-3.5" /> Duplicate

   400→            onClick={() => { toggleLock(contextMenu.id); setContextMenu(null); }}>

   407→          </button>

   409→      )}
   410→
   411→      {/* Properties panel — stays visible while module is selected */}
   412→      {selectedMod && onUpdateModules && (

   418→            <div className="ml-auto flex gap-1">
   419→              <button onClick={() => duplicateModule(selectedMod.id)} className="flex items-center gap-1 rounded border border-border px-2 py-1 text-xs hover:bg-muted" title="Duplicate">
   420→                <Copy className="h-3 w-3" /> Duplicate
   421→              </button>
   422→              <button onClick={() => toggleLock(selectedMod.id)} className="flex items-center gap-1 rounded border border-border px-2 py-1 text-xs hover:bg-muted" title="Lock/Unlock">
   423→                {lockedIds.has(selectedMod.id) ? <Unlock className="h-3 w-3" /> : <Lock className="h-3 w-3" />}
   424→              </button>
   425→              <button onClick={() => deleteModule(selectedMod.id)} className="flex items-center gap-1 rounded bg-red-500 px-2 py-1 text-xs text-white hover:bg-red-600">
   426→                <Trash2 className="h-3 w-3" /> Delete
   427→              </button>
   428→            </div>
   429→          </div>
   430→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-4 lg:grid-cols-6">
   431→            <div className="space-y-1">
   432→              <Label className="text-[10px] text-muted-foreground">Width (mm)</Label>
   433→              <Input type="number" min="100" step="50" value={selectedMod.width} disabled={lockedIds.has(selectedMod.id)}
   434→                onChange={(e) => onUpdateModules(modules.map(m => m.id === selectedMod.id ? { ...m, width: Math.max(100, Number(e.target.value) || 100) } : m))}
   435→                className="h-8 text-xs" />
   436→            </div>
   437→            <div className="space-y-1">
   438→              <Label className="text-[10px] text-muted-foreground">Height (mm)</Label>
   439→              <Input type="number" min="100" step="50" value={selectedMod.height} disabled={lockedIds.has(selectedMod.id)}
   440→                onChange={(e) => onUpdateModules(modules.map(m => m.id === selectedMod.id ? { ...m, height: Math.max(100, Number(e.target.value) || 100) } : m))}
   441→                className="h-8 text-xs" />
   442→            </div>
   443→            <div className="space-y-1">
   444→              <Label className="text-[10px] text-muted-foreground">X Position</Label>

   446→                onChange={(e) => onUpdateModules(modules.map(m => m.id === selectedMod.id ? { ...m, x: Math.max(0, Number(e.target.value) || 0) } : m))}

   457→              <Select value={selectedMod.type} disabled={lockedIds.has(selectedMod.id)}
   458→                onValueChange={(v) => onUpdateModules(modules.map(m => m.id === selectedMod.id ? { ...m, type: v as any } : m))}>
   459→                <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   460→                <SelectContent>{CABINET_TYPES.map(t => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}</SelectContent>

   467→                <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>

   470→            </div>
   471→          </div>
   472→          <p className="mt-2 text-[10px] text-muted-foreground">
   473→            Drag to move · 8 amber nodes to resize (corners + edges) · Right-click for menu · Edges snap to nearby modules
   474→          </p>
   475→        </div>
   476→      )}
   477→    </div>
