   131→function num(v: unknown): number {
   132→  const n = typeof v === "string" ? parseFloat(v) : Number(v);
   133→  return Number.isFinite(n) ? n : 0;
   134→}

   136→function fmtDate(s?: string | null): string {
   137→  if (!s) return "—";
   138→  try {
   139→    return new Date(s).toLocaleDateString(undefined, {
   140→      year: "numeric",
   141→      month: "short",
   142→      day: "numeric",
   143→    });
   144→  } catch {
   145→    return "—";
   146→  }
   147→}
   148→
   149→function fmtDateTime(s?: string | null): string {
   150→  if (!s) return "—";
   151→  try {
   152→    return new Date(s).toLocaleString(undefined, {
   153→      month: "short",
   154→      day: "numeric",
   155→      hour: "2-digit",
   156→      minute: "2-digit",
   157→    });
   158→  } catch {
   159→    return "—";
   160→  }
   161→}

   331→function DashboardTab() {
   332→  const { data: inv } = useQuery({ queryKey: ["inventory"], queryFn: inventoryApi.list });
   333→  const { data: wh } = useQuery({ queryKey: ["warehouses"], queryFn: warehousesApi.list });
   334→  const { data: reqs } = useQuery({
   335→    queryKey: ["stock-requests", "list"],
   336→    queryFn: () => stockRequestsApi.list(),
   337→  });
   338→  const { data: tr } = useQuery({
   339→    queryKey: ["stock-transfers", "list"],
   340→    queryFn: () => stockTransfersApi.list(),
   341→  });
   342→  const { data: issues } = useQuery({
   343→    queryKey: ["goods-issues", "list"],
   344→    queryFn: () => goodsIssuesApi.list(),
   345→  });
   346→  const { data: returns } = useQuery({
   347→    queryKey: ["goods-returns", "list"],
   348→    queryFn: () => goodsReturnsApi.list(),
   349→  });
   350→  const { data: adj } = useQuery({
   351→    queryKey: ["stock-adjustments", "list"],
   352→    queryFn: () => stockAdjustmentsApi.list(),
   353→  });
   354→
   355→  const items = inv?.items ?? [];
   356→  const warehouses = wh?.warehouses ?? [];
   357→  const requests = reqs?.requests ?? [];
   358→  const transfers = tr?.transfers ?? [];
   359→  const allIssues = issues?.issues ?? [];
   360→  const allReturns = returns?.returns ?? [];
   361→  const adjustments = adj?.adjustments ?? [];
   362→
   363→  const lowStock = items.filter(
   364→    (i) => num(i.reorderPoint) > 0 && num(i.stockLevel) <= num(i.reorderPoint) && i.status === "active"
   365→  );
   366→  const pendingRequests = requests.filter((r) => r.status === "pending");
   367→  const pendingTransfers = transfers.filter((t) => t.status === "in_transit" || t.status === "draft");
   368→
   369→  type Movement = { date: string; type: string; number: string; description: string };
   370→  const movements: Movement[] = [
   371→    ...allIssues.map((i) => ({
   372→      date: i.createdAt,
   373→      type: "Issue",
   374→      number: i.issueNo,
   375→      description: `${i.lines.length} item(s) issued to ${i.job?.orderNumber ?? "—"}`,
   376→    })),
   377→    ...allReturns.map((r) => ({
   378→      date: r.createdAt,
   379→      type: "Return",
   380→      number: r.returnNo,
   381→      description: `${r.lines.length} item(s) returned from ${r.job?.orderNumber ?? "—"}`,
   382→    })),
   383→    ...adjustments.map((a) => ({
   384→      date: a.createdAt,
   385→      type: "Adjustment",
   386→      number: a.adjNo,
   387→      description: `${a.item?.name ?? "—"} (${a.type}, ${num(a.diff) >= 0 ? "+" : ""}${num(a.diff)})`,
   388→    })),
   389→    ...transfers
   390→      .filter((t) => t.status !== "draft")
   391→      .map((t) => ({
   392→        date: t.createdAt,
   393→        type: "Transfer",
   394→        number: t.transferNo,
   395→        description: `${t.fromWarehouse?.code ?? "—"} → ${t.toWarehouse?.code ?? "—"}`,
   396→      })),
   397→  ].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()).slice(0, 5);
   398→
   399→  return (
   400→    <div className="space-y-5">
   401→      <TabHeader title="Dashboard" description="Overview of inventory health and pending work." />
   402→
   403→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-3 xl:grid-cols-6">
   404→        <MiniStat label="Total Items" value={items.length} icon={Package} tint="bg-primary/10 text-primary" />
   405→        <MiniStat label="Warehouses" value={warehouses.length} icon={WarehouseIcon} tint="bg-teal-500/10 text-teal-600" />
   406→        <MiniStat label="Low Stock" value={lowStock.length} icon={TrendingDown} tint="bg-rose-500/10 text-rose-600" />
   407→        <MiniStat label="Pending Requests" value={pendingRequests.length} icon={ClipboardList} tint="bg-amber-500/10 text-amber-600" />
   408→        <MiniStat label="Pending Transfers" value={pendingTransfers.length} icon={ArrowLeftRight} tint="bg-violet-500/10 text-violet-600" />
   409→        <MiniStat label="Active Items" value={items.filter((i) => i.status === "active").length} icon={CheckCircle2} tint="bg-emerald-500/10 text-emerald-600" />
   410→      </div>
   411→
   412→      <div className="grid grid-cols-1 gap-5 lg:grid-cols-2">
   413→        <Card>
   414→          <CardHeader className="pb-3">
   415→            <CardTitle className="flex items-center gap-2 text-base">
   416→              <Clock className="h-4 w-4 text-primary" />
   417→              Recent Activity
   418→            </CardTitle>
   419→            <CardDescription>Latest stock movements across all warehouses</CardDescription>
   420→          </CardHeader>
   421→          <CardContent className="p-0">
   422→            {movements.length === 0 ? (
   423→              <EmptyState icon={Activity} title="No recent activity" description="Movements will appear here once you start issuing or transferring stock." />
   424→            ) : (
   425→              <div className="divide-y">
   426→                {movements.map((m, idx) => (
   427→                  <div key={idx} className="flex items-start gap-3 px-4 py-3">
   428→                    <div className="rounded-md bg-muted px-2 py-1 text-xs font-medium">
   429→                      {m.type}
   430→                    </div>
   431→                    <div className="min-w-0 flex-1">
   432→                      <p className="text-sm font-medium truncate">{m.number}</p>
   433→                      <p className="text-xs text-muted-foreground truncate">{m.description}</p>
   434→                    </div>
   435→                    <div className="text-xs text-muted-foreground whitespace-nowrap">{fmtDate(m.date)}</div>
   436→                  </div>
   437→                ))}
   438→              </div>
   439→            )}
   440→          </CardContent>
   441→        </Card>
   442→
   443→        <Card>
   444→          <CardHeader className="pb-3">
   445→            <CardTitle className="flex items-center gap-2 text-base">
   446→              <AlertTriangle className="h-4 w-4 text-rose-600" />
   447→              Low Stock Alerts
   448→            </CardTitle>
   449→            <CardDescription>Items at or below reorder point</CardDescription>
   450→          </CardHeader>
   451→          <CardContent className="p-0">
   452→            {lowStock.length === 0 ? (
   453→              <EmptyState icon={CheckCircle2} title="All stocked up" description="No items are below their reorder point." />
   454→            ) : (
   455→              <div className="max-h-72 overflow-y-auto divide-y">
   456→                {lowStock.slice(0, 8).map((i) => {
   457→                  const stock = num(i.stockLevel);
   458→                  const reorder = num(i.reorderPoint);
   459→                  const isOut = stock === 0;
   460→                  return (
   461→                    <div key={i.id} className="flex items-center justify-between gap-3 px-4 py-3">
   462→                      <div className="min-w-0">
   463→                        <p className="text-sm font-medium truncate">{i.name}</p>
   464→                        <p className="text-xs text-muted-foreground">
   465→                          {i.code ?? i.material} · {i.unit}
   466→                        </p>
   467→                      </div>
   468→                      <div className="text-right">
   469→                        <p className={`text-sm font-bold tabular-nums ${isOut ? "text-rose-600" : "text-amber-600"}`}>
   470→                          {stock} {i.unit}
   471→                        </p>
   472→                        <p className="text-xs text-muted-foreground">reorder @ {reorder}</p>
   473→                      </div>
   474→                    </div>
   475→                  );
   476→                })}
   477→              </div>
   478→            )}
   479→          </CardContent>
   480→        </Card>
   481→      </div>
   482→    </div>
   483→  );
   484→}

   487→function ItemsTab() {

   491→  const [categoryFilter, setCategoryFilter] = React.useState("all");

   493→  const [editing, setEditing] = React.useState<InventoryItem | null>(null);
   494→  const [createOpen, setCreateOpen] = React.useState(false);
   495→  const [deleting, setDeleting] = React.useState<InventoryItem | null>(null);
   496→
   497→  const { data, isLoading } = useQuery({ queryKey: ["inventory"], queryFn: inventoryApi.list });
   498→  const { data: catData } = useQuery({ queryKey: ["inventory-categories"], queryFn: inventoryCategoriesApi.list });
   499→  const items = data?.items ?? [];
   500→  const categories = catData?.categories ?? [];
   501→
   502→  const filtered = React.useMemo(() => {
   503→    return items.filter((i) => {
   504→      const q = search.toLowerCase();
   505→      const matchesSearch =
   506→        !q ||
   507→        i.name.toLowerCase().includes(q) ||
   508→        (i.code ?? "").toLowerCase().includes(q) ||
   509→        (i.supplier ?? "").toLowerCase().includes(q);
   510→      const matchesMaterial = materialFilter === "all" || i.material === materialFilter;
   511→      const matchesCat = categoryFilter === "all" || i.categoryId === categoryFilter;
   512→      const matchesStatus = statusFilter === "all" || i.status === statusFilter;
   513→      return matchesSearch && matchesMaterial && matchesCat && matchesStatus;
   514→    });
   515→  }, [items, search, materialFilter, categoryFilter, statusFilter]);
   516→
   517→  const deleteMutation = useMutation({
   518→    mutationFn: (id: string) => inventoryApi.remove(id),
   519→    onSuccess: () => {
   520→      toast.success("Item deleted");
   521→      queryClient.invalidateQueries({ queryKey: ["inventory"] });
   522→      setDeleting(null);
   523→    },
   524→    onError: (e: Error) => toast.error("Delete failed", { description: e.message }),
   525→  });
   526→
   527→  function exportCsv() {
   528→    const headers = ["Name", "Code", "Material", "Thickness", "Unit", "Stock", "Min", "Reorder", "Supplier", "Category", "Status"];
   529→    const rows = filtered.map((i) => ({
   530→      Name: i.name,

   574→  function importCsv(file: File) {
   575→    const reader = new FileReader();
   576→    reader.onload = async () => {
   577→      const text = String(reader.result ?? "");
   578→      const lines = text.split("\n").filter((l) => l.trim());
   579→      if (lines.length < 2) {
   580→        toast.error("CSV must have a header and at least one row");
   581→        return;
   582→      }
   583→      const headers = lines[0].split(",").map((h) => h.trim().toLowerCase());
   584→      let imported = 0;
   585→      for (const line of lines.slice(1)) {
   586→        const cells = line.split(",").map((c) => c.trim().replace(/^"|"$/g, ""));
   587→        const get = (key: string) => {
   588→          const idx = headers.indexOf(key);
   589→          return idx >= 0 ? cells[idx] : "";
   590→        };
   591→        const name = get("name");
   592→        const material = get("material") || "Other";
   593→        if (!name) continue;
   594→        try {
   595→          await inventoryApi.create({
   596→            name,
   597→            code: get("code") || null,
   598→            material,
   599→            thickness: get("thickness") || null,
   600→            unit: get("unit") || "sheet",
   601→            stockLevel: Number(get("stock")) || 0,
   602→            minStock: Number(get("min")) || 0,
   603→            reorderPoint: Number(get("reorder")) || 0,
   604→            supplier: get("supplier") || null,
   605→            status: "active",
   606→          });
   607→          imported++;
   608→        } catch (e) {
   609→          // skip duplicates
   610→          console.error("Import error:", e);
   611→        }
   612→      }
   613→      toast.success(`Imported ${imported} item(s)`);
   614→      queryClient.invalidateQueries({ queryKey: ["inventory"] });
   615→    };
   616→    reader.readAsText(file);
   617→  }
