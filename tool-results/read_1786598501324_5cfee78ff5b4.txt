   184→function num(v: unknown): number {

   189→function fmtDate(s?: string | null): string {
   190→  if (!s) return "—";
   191→  try {
   192→    return new Date(s).toLocaleDateString(undefined, {
   193→      year: "numeric",
   194→      month: "short",
   195→      day: "numeric",
   196→    });
   197→  } catch {
   198→    return "—";
   199→  }
   200→}
   201→
   202→function fmtDateTime(s?: string | null): string {
   203→  if (!s) return "—";
   204→  try {
   205→    return new Date(s).toLocaleString(undefined, {
   206→      month: "short",
   207→      day: "numeric",
   208→      hour: "2-digit",
   209→      minute: "2-digit",
   210→    });
   211→  } catch {
   212→    return "—";
   213→  }
   214→}
   215→
   216→function toInputDate(s?: string | null): string {
   217→  if (!s) return "";
   218→  try {
   219→    const d = new Date(s);
   220→    if (Number.isNaN(d.getTime())) return "";
   221→    return d.toISOString().slice(0, 10);
   222→  } catch {
   223→    return "";
   224→  }
   225→}
   226→
   227→function toInputDateTimeLocal(s?: string | null): string {
   228→  if (!s) return "";
   229→  try {
   230→    const d = new Date(s);
   231→    if (Number.isNaN(d.getTime())) return "";
   232→    // local time
   233→    const offsetMs = d.getTimezoneOffset() * 60000;
   234→    return new Date(d.getTime() - offsetMs).toISOString().slice(0, 16);
   235→  } catch {
   236→    return "";
   237→  }
   238→}

   910→function TimeLogDialog({
   911→  jobId,
   912→  log,
   913→  users,
   914→  open,
   915→  onOpenChange,
   916→}: {
   917→  jobId: string;
   918→  log: JobTimeLog | null;
   919→  users: { id: string; fullName: string; role: string }[];
   920→  open: boolean;
   921→  onOpenChange: (o: boolean) => void;
   922→}) {
   923→  const queryClient = useQueryClient();
   924→  const [workerName, setWorkerName] = React.useState("");
   925→  const [userId, setUserId] = React.useState("");
   926→  const [workDate, setWorkDate] = React.useState("");
   927→  const [workType, setWorkType] = React.useState("onsite");
   928→  const [clockIn, setClockIn] = React.useState("");
   929→  const [clockOut, setClockOut] = React.useState("");

   932→  const [useClock, setUseClock] = React.useState(true);
   933→
   934→  React.useEffect(() => {
   935→    if (log) {
   936→      setWorkerName(log.workerName);
   937→      setUserId(log.userId ?? "");
   938→      setWorkDate(toInputDate(log.workDate));
   939→      setWorkType(log.workType);
   940→      setClockIn(toInputDateTimeLocal(log.clockIn));
   941→      setClockOut(toInputDateTimeLocal(log.clockOut));
   942→      setHoursWorked(log.clockIn && log.clockOut ? "" : String(log.hoursWorked));
   943→      setNotes(log.notes ?? "");

   979→        payload.hoursWorked = Number(hoursWorked);
   980→      }

   982→        return jobTimeLogsApi.update(log.id, payload);

   988→      queryClient.invalidateQueries({ queryKey: ["job-time-logs", jobId] });
   989→      queryClient.invalidateQueries({ queryKey: ["job-insights", jobId] });

  1013→              toast.error("Worker name is required");
  1014→              return;
  1015→            }
  1016→            save.mutate();
  1017→          }}
  1018→          className="space-y-3"
  1019→        >
  1020→          <div className="grid grid-cols-2 gap-3">
  1021→            <div className="space-y-1.5">
  1022→              <Label className="text-xs">Worker (system user)</Label>
