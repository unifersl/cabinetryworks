   538→// Time tracking — per person, per day, multiple sessions per day (factory / onsite / travel)
   539→model JobTimeLog {
   540→  id          String   @id
   541→  jobId       String

   543→  workerName  String   // free-text worker name (always populated for display)

   547→  clockOut    DateTime?
   548→  hoursWorked Decimal  @default(0) // computed hours (decimal allows 1.5h etc.)
   549→  hourlyRate  Decimal  @default(0) // labor rate per hour (LKR)
   550→  laborCost   Decimal  @default(0) // computed: hoursWorked × hourlyRate
   551→  notes       String?
   552→  createdBy   String?
   553→  createdAt   DateTime @default(now())
   554→  updatedAt   DateTime @updatedAt
   555→
   556→  job  JobOrder @relation(fields: [jobId], references: [id], onDelete: Cascade)
   557→  user User?    @relation(fields: [userId], references: [id], onDelete: SetNull)
   558→
   559→  @@index([jobId])
   560→  @@index([workDate])
   561→  @@index([userId])
