   111→   111→function timeAgo(iso: string): string {
   112→   112→  const diff = Date.now() - new Date(iso).getTime();
   113→   113→  const mins = Math.floor(diff / 60000);
   114→   114→  if (mins < 1) return "just now";
   115→   115→  if (mins < 60) return `${mins}m ago`;
   116→   116→  const hours = Math.floor(mins / 60);
   117→   117→  if (hours < 24) return `${hours}h ago`;
   118→   118→  const days = Math.floor(hours / 24);
   119→   119→  if (days < 7) return `${days}d ago`;
   120→   120→  return new Date(iso).toLocaleDateString(undefined, { month: "short", day: "numeric" });
   121→   121→}
