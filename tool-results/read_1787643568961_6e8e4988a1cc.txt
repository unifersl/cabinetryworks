   123→   123→function timeAgo(iso: string): string {
   124→   124→  const diff = Date.now() - new Date(iso).getTime();
   125→   125→  const mins = Math.floor(diff / 60000);
   126→   126→  if (mins < 1) return "just now";
   127→   127→  if (mins < 60) return `${mins}m ago`;
   128→   128→  const hours = Math.floor(mins / 60);
   129→   129→  if (hours < 24) return `${hours}h ago`;
   130→   130→  const days = Math.floor(hours / 24);
   131→   131→  if (days < 7) return `${days}d ago`;
   132→   132→  return new Date(iso).toLocaleDateString(undefined, { month: "short", day: "numeric" });
   133→   133→}
   134→   134→
   135→   135→function greetingFor(date: Date): string {
   136→   136→  const h = date.getHours();
   137→   137→  if (h < 12) return "Good morning";
   138→   138→  if (h < 17) return "Good afternoon";
   139→   139→  if (h < 21) return "Good evening";
   140→   140→  return "Working late";
   141→   141→}

   143→   143→/** Returns "Today" / "Yesterday" / "Aug 19" (short locale). */
   144→   144→function dayBucket(iso: string): string {
   145→   145→  const d = new Date(iso);
   146→   146→  const now = new Date();
   147→   147→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   148→   148→  const startOfDay = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   149→   149→  const diffDays = Math.round(
   150→   150→    (startOfToday.getTime() - startOfDay.getTime()) / 86400000
   151→   151→  );
   152→   152→  if (diffDays === 0) return "Today";
   153→   153→  if (diffDays === 1) return "Yesterday";
   154→   154→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
   155→   155→}

   157→   157→type ActivityFilter = "all" | "jobs" | "measurements" | "cutting";
   158→   158→
   159→   159→function filterActivity(
   160→   160→  events: ActivityEvent[],
   161→   161→  filter: ActivityFilter
   162→   162→): ActivityEvent[] {
   163→   163→  if (filter === "all") return events;
   164→   164→  if (filter === "jobs")
   165→   165→    return events.filter((e) =>
   166→   166→      ["job_created", "job_assigned", "job_status"].includes(e.type)
   167→   167→    );
   168→   168→  if (filter === "measurements")
   169→   169→    return events.filter((e) => e.type === "measurement");
   170→   170→  return events.filter((e) => e.type === "cutting_list");
   171→   171→}
