    15→async function request<T>(
    16→  url: string,
    17→  options?: RequestInit
    18→): Promise<T> {
    19→  let lastError: Error | null = null;
    20→  const maxRetries = 2;
    21→
    22→  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    23→    try {
    24→      const controller = new AbortController();
    25→      const timeoutId = setTimeout(() => controller.abort(), 15000);
    26→
    27→      const res = await fetch(url, {
    28→        ...options,
    29→        headers: {
    30→          "Content-Type": "application/json",
    31→          ...(options?.headers || {}),
    32→        },
    33→        credentials: "same-origin",
    34→        signal: controller.signal,
    35→      });
    36→
    37→      clearTimeout(timeoutId);
    38→
    39→      let data: unknown = null;
    40→      try {
    41→        data = await res.json();
    42→      } catch {
    43→        /* non-json response */
    44→      }
    45→
    46→      if (!res.ok) {
    47→        const msg =
    48→          (data && typeof data === "object" && "error" in data
    49→            ? String((data as Record<string, unknown>).error)
    50→            : null) ?? `Request failed (${res.status})`;
    51→        const err = new Error(msg) as Error & {
    52→          status: number;
    53→          fallback?: string;
    54→          detail?: string;
    55→        };
    56→        err.status = res.status;
    57→        if (data && typeof data === "object") {
    58→          const d = data as Record<string, unknown>;
    59→          if (typeof d.fallback === "string") err.fallback = d.fallback;
    60→          if (typeof d.detail === "string") err.detail = d.detail;
    61→        }
    62→        throw err;
    63→      }
    64→      return data as T;
    65→    } catch (error) {
    66→      lastError = error as Error;
    67→      // Retry on network errors (502, connection refused, abort/timeout)
    68→      // but not on 4xx client errors
    69→      const status = (error as { status?: number }).status;
    70→      const isNetworkError =
    71→        !status ||
    72→        status === 502 ||
    73→        status === 503 ||
    74→        error instanceof TypeError ||
    75→        (error as Error).name === "AbortError";
    76→
    77→      if (isNetworkError && attempt < maxRetries) {
    78→        // Exponential backoff: 500ms, 1000ms
    79→        await new Promise((r) => setTimeout(r, 500 * (attempt + 1)));
    80→        continue;
    81→      }
    82→      throw error;
    83→    }
    84→  }
    85→
    86→  throw lastError ?? new Error("Request failed after retries");
    87→}
