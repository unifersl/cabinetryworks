    69→    69→export function ElevationCanvas({ data, height = 500, onUpdateModules }: ElevationCanvasProps) {
    70→    70→  const { modules: rawModules, overallDimensions } = data;
    71→    71→  // Guard overall dimensions against NaN/undefined — a common cause of NaN SVG attributes
    72→    72→  const totalW = safeNum(overallDimensions?.width, 3400);
    73→    73→  const totalH = safeNum(overallDimensions?.height, 2150);
    74→    74→
    75→    75→  const [zoom, setZoom] = React.useState(1);
    76→    76→  const [selectedId, setSelectedId] = React.useState<string | null>(null);
    77→    77→  const [lockedIds, setLockedIds] = React.useState<Set<string>>(new Set());
    78→    78→  const [contextMenu, setContextMenu] = React.useState<{ id: string; x: number; y: number } | null>(null);
    79→    79→  const [isFullscreen, setIsFullscreen] = React.useState(false);
    80→    80→  const [showHelp, setShowHelp] = React.useState(false);
    81→    81→
    82→    82→  const dragData = React.useRef<{
    83→    83→    id: string; type: "move" | "resize"; edge?: string;
    84→    84→    startPxX: number; startPxY: number;
    85→    85→    origX: number; origY: number; origW: number; origH: number;
    86→    86→  } | null>(null);
    87→    87→  const innerRef = React.useRef<HTMLDivElement>(null);
    88→    88→
    89→    89→  // Normalize legacy modules so drawerHeights / shelfHeights always exist AND
    90→    90→  // all core numeric fields are finite numbers. This is the primary guard
    91→    91→  // against the "Received NaN for attribute" console errors — it migrates any
    92→    92→  // module loaded from localStorage that was saved before these fields existed
    93→    93→  // or that has undefined/NaN width/height/x/y.
    94→    94→  const modules = React.useMemo(
    95→    95→    () => rawModules.map((m) => ensureDrawersShelves(m)),
    96→    96→    [rawModules],
    97→    97→  );
    98→    98→
    99→    99→  const basePxPerMm = Math.min(700 / Math.max(totalW, 1), 350 / Math.max(totalH, 1));
   100→   100→  const pxPerMm = safeNum(basePxPerMm * zoom, 0.2);
   101→   101→  const wallPxW = totalW * pxPerMm;
   102→   102→  const wallPxH = totalH * pxPerMm;
   103→   103→  const PAD = 40;
   104→   104→  const innerW = Math.max(wallPxW + PAD * 2, 700);
   105→   105→  const innerH = Math.max(wallPxH + PAD * 2, 400);
   106→   106→  // NaN-safe mm→px converter: any non-finite input falls back to 0
   107→   107→  const mm = (v: number) => safeNum(v, 0) * pxPerMm;
   108→   108→  const SNAP_THRESHOLD_MM = 50;
   109→   109→
   110→   110→  // --- Accidental close prevention ---
   111→   111→  // When there are modules on the canvas, warn before the tab/window is closed
   112→   112→  // or refreshed so unsaved work isn't lost.
   113→   113→  React.useEffect(() => {
   114→   114→    if (modules.length === 0) return;
   115→   115→    function onBeforeUnload(e: BeforeUnloadEvent) {
   116→   116→      e.preventDefault();
   117→   117→      e.returnValue = "";
   118→   118→    }
   119→   119→    window.addEventListener("beforeunload", onBeforeUnload);
   120→   120→    return () => window.removeEventListener("beforeunload", onBeforeUnload);
   121→   121→  }, [modules.length]);
   122→   122→
   123→   123→  // Esc to exit fullscreen
   124→   124→  React.useEffect(() => {
   125→   125→    if (!isFullscreen) return;

   127→   127→      if (e.key === "Escape") setIsFullscreen(false);
   128→   128→    }
   129→   129→    window.addEventListener("keydown", onKey);
   130→   130→    return () => window.removeEventListener("keydown", onKey);
   131→   131→  }, [isFullscreen]);

   133→   133→  // --- Snap helper: find nearby edges to snap to ---
   134→   134→  function getSnappedPos(modId: string, newX: number, newY: number, newW: number, newH: number): { x: number; y: number } {
   135→   135→    let snapX = newX, snapY = newY;
   136→   136→    const myEdges = {
   137→   137→      left: newX, right: newX + newW,
   138→   138→      top: newY, bottom: newY + newH,
   139→   139→      centerX: newX + newW / 2, centerY: newY + newH / 2,
   140→   140→    };
   141→   141→    for (const other of modules) {
   142→   142→      if (other.id === modId) continue;
   143→   143→      const oEdges = {
   144→   144→        left: other.x, right: other.x + other.width,
   145→   145→        top: other.y, bottom: other.y + other.height,
   146→   146→        centerX: other.x + other.width / 2, centerY: other.y + other.height / 2,
   147→   147→      };
   148→   148→      for (const [myKey, myVal] of Object.entries(myEdges)) {
   149→   149→        if (!myKey.includes("x") && myKey !== "left" && myKey !== "right" && myKey !== "centerX") continue;
   150→   150→        for (const [oKey, oVal] of Object.entries(oEdges)) {
   151→   151→          if (!oKey.includes("x") && oKey !== "left" && oKey !== "right" && oKey !== "centerX") continue;
   152→   152→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
   153→   153→            if (myKey === "left" || myKey === "right" || myKey === "centerX") {
   154→   154→              const diff = oVal - myVal;
   155→   155→              snapX = newX + diff;
   156→   156→            }
   157→   157→          }
   158→   158→        }
   159→   159→      }
   160→   160→      for (const [myKey, myVal] of Object.entries(myEdges)) {
   161→   161→        if (myKey !== "top" && myKey !== "bottom" && myKey !== "centerY") continue;
   162→   162→        for (const [oKey, oVal] of Object.entries(oEdges)) {
   163→   163→          if (oKey !== "top" && oKey !== "bottom" && oKey !== "centerY") continue;
   164→   164→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
   165→   165→            if (myKey === "top" || myKey === "bottom" || myKey === "centerY") {
   166→   166→              const diff = oVal - myVal;
   167→   167→              snapY = newY + diff;
   168→   168→            }
   169→   169→          }
   170→   170→        }
   171→   171→      }
   172→   172→    }
   173→   173→    if (Math.abs(snapX) < SNAP_THRESHOLD_MM) snapX = 0;
   174→   174→    if (Math.abs(snapY) < SNAP_THRESHOLD_MM) snapY = 0;
   175→   175→    if (Math.abs(snapX + newW - totalW) < SNAP_THRESHOLD_MM) snapX = totalW - newW;
   176→   176→    if (Math.abs(snapY + newH - totalH) < SNAP_THRESHOLD_MM) snapY = totalH - newH;
   177→   177→    return { x: snapX, y: snapY };
   178→   178→  }

   180→   180→  function handleDrop(e: React.DragEvent) {
   181→   181→    e.preventDefault();
   182→   182→    const type = e.dataTransfer.getData("text/cabinet-type");
   183→   183→    if (!type || !onUpdateModules) return;
   184→   184→    const item = PALETTE.find((p) => p.type === type);
   185→   185→    if (!item) return;
   186→   186→    const inner = innerRef.current;
   187→   187→    if (!inner) return;
   188→   188→    const rect = inner.getBoundingClientRect();
   189→   189→    const mmX = (e.clientX - rect.left - PAD) / pxPerMm;
   190→   190→    const mmY = (e.clientY - rect.top - PAD) / pxPerMm;
   191→   191→    const newCode = String(modules.length + 1).padStart(2, "0");
   192→   192→    const newMod = createEmptyModule(newCode);
   193→   193→    newMod.type = type as any;
   194→   194→    newMod.width = item.w;
   195→   195→    newMod.height = item.h;
   196→   196→    newMod.x = Math.round(Math.max(0, Math.min(totalW - item.w, mmX - item.w / 2)));
   197→   197→    newMod.y = Math.round(Math.max(0, Math.min(totalH - item.h, mmY - item.h / 2)));
   198→   198→    onUpdateModules([...modules, newMod]);
   199→   199→    setSelectedId(newMod.id);
   200→   200→  }

   202→   202→  function onModuleDown(e: React.PointerEvent, mod: CabinetModule) {
   203→   203→    if (!onUpdateModules) return;
   204→   204→    e.stopPropagation();
   205→   205→    setSelectedId(mod.id);
   206→   206→    if (lockedIds.has(mod.id)) return;
   207→   207→    dragData.current = {
   208→   208→      id: mod.id, type: "move",
   209→   209→      startPxX: e.clientX, startPxY: e.clientY,
   210→   210→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   211→   211→    };
   212→   212→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   213→   213→  }

   216→   216→    e.stopPropagation();
   217→   217→    setSelectedId(mod.id);
   218→   218→    if (lockedIds.has(mod.id)) return;
   219→   219→    dragData.current = {
   220→   220→      id: mod.id, type: "resize", edge,
   221→   221→      startPxX: e.clientX, startPxY: e.clientY,
   222→   222→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   223→   223→    };

   225→   225→  }

   227→   227→  function onCanvasMove(e: React.PointerEvent) {
   228→   228→    const dd = dragData.current;
   229→   229→    if (!dd || !onUpdateModules) return;
   230→   230→    const dxMm = (e.clientX - dd.startPxX) / pxPerMm;
   231→   231→    const dyMm = (e.clientY - dd.startPxY) / pxPerMm;
   232→   232→    if (dd.type === "move") {
   233→   233→      let newX = Math.max(0, Math.min(totalW - 50, Math.round(dd.origX + dxMm)));
   234→   234→      let newY = Math.max(0, Math.min(totalH - 50, Math.round(dd.origY + dyMm)));
   235→   235→      const snapped = getSnappedPos(dd.id, newX, newY, dd.origW, dd.origH);
   236→   236→      newX = snapped.x; newY = snapped.y;
   237→   237→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, x: newX, y: newY } : m));
   238→   238→    } else if (dd.type === "resize") {
   239→   239→      const min = 100;
   240→   240→      let nw = dd.origW, nh = dd.origH, nx = dd.origX, ny = dd.origY;
   241→   241→      if (dd.edge === "right") nw = Math.max(min, Math.round(dd.origW + dxMm));
   242→   242→      if (dd.edge === "left") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); }
   243→   243→      if (dd.edge === "bottom") nh = Math.max(min, Math.round(dd.origH + dyMm));
   244→   244→      if (dd.edge === "top") { nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   245→   245→      if (dd.edge === "br") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   246→   246→      if (dd.edge === "bl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   247→   247→      if (dd.edge === "tr") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   248→   248→      if (dd.edge === "tl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   249→   249→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, width: nw, height: nh, x: nx, y: ny } : m));
   250→   250→    }
   251→   251→  }

   255→   255→  function deleteModule(id: string) {
   256→   256→    if (!onUpdateModules) return;
   257→   257→    onUpdateModules(modules.filter((m) => m.id !== id).map((m, i) => ({ ...m, code: String(i + 1).padStart(2, "0") })));
   258→   258→    setSelectedId(null);
   259→   259→    setLockedIds((s) => { const n = new Set(s); n.delete(id); return n; });
   260→   260→  }

   262→   262→  function duplicateModule(id: string) {
   263→   263→    if (!onUpdateModules) return;
   264→   264→    const mod = modules.find((m) => m.id === id);
   265→   265→    if (!mod) return;
   266→   266→    const newCode = String(modules.length + 1).padStart(2, "0");
   267→   267→    const copy: CabinetModule = {
   268→   268→      ...mod,
   269→   269→      id: crypto.randomUUID(),
   270→   270→      code: newCode,
   271→   271→      x: mod.x + 50,
   272→   272→      y: mod.y + 50,
   273→   273→      cuttingList: mod.cuttingList.map(c => ({ ...c, id: crypto.randomUUID() })),
   274→   274→      drawerHeights: [...(mod.drawerHeights ?? [])],
   275→   275→      shelfHeights: [...(mod.shelfHeights ?? [])],
   276→   276→    };
   277→   277→    onUpdateModules([...modules, copy]);
   278→   278→    setSelectedId(copy.id);
   279→   279→    toast.success(`Duplicated Module ${mod.code} → ${newCode}`);
   280→   280→  }

   292→   292→  function setDrawerCount(id: string, count: number) {
   293→   293→    const c = Math.max(1, Math.min(10, Math.floor(count)));
   294→   294→    const mod = modules.find((m) => m.id === id);
   295→   295→    if (!mod) return;
   296→   296→    patchModule(id, { drawerHeights: distributeEvenly(c, mod.height, 0) });
   297→   297→  }

   299→   299→  function setDrawerHeight(id: string, idx: number, h: number) {
   300→   300→    const mod = modules.find((m) => m.id === id);
   301→   301→    if (!mod) return;
   302→   302→    const next = [...(mod.drawerHeights ?? [])];
   303→   303→    next[idx] = Math.max(50, Math.round(h));
   304→   304→    patchModule(id, { drawerHeights: next });
   305→   305→  }

   307→   307→  function setShelfCount(id: string, count: number) {
   308→   308→    const c = Math.max(0, Math.min(8, Math.floor(count)));
   309→   309→    const mod = modules.find((m) => m.id === id);
   310→   310→    if (!mod) return;
   311→   311→    const positions = Array.from({ length: c }, (_, i) =>
   312→   312→      Math.round((mod.height * (i + 1)) / (c + 1)),
   313→   313→    );
   314→   314→    patchModule(id, { shelfHeights: positions });
   315→   315→  }

   317→   317→  function setShelfHeight(id: string, idx: number, hFromBottom: number) {
   318→   318→    const mod = modules.find((m) => m.id === id);
   319→   319→    if (!mod) return;
   320→   320→    const next = [...(mod.shelfHeights ?? [])];
   321→   321→    next[idx] = Math.max(20, Math.min(mod.height - 20, Math.round(hFromBottom)));
   322→   322→    patchModule(id, { shelfHeights: next });
   323→   323→  }

   325→   325→  function onModuleContextMenu(e: React.MouseEvent, mod: CabinetModule) {
