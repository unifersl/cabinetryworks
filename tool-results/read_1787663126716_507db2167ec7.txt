    86→// Sheet area = 2440mm × 1220mm = 2.977 m² (matches the Site Notebook estimator)
    87→const STANDARD_SHEET_AREA_SQM = (2440 * 1220) / 1_000_000;
    88→
    89→/** Compute the sheets required for a given total area in m² (with 15% wastage). */
    90→function sheetsNeededFor(areaSqm: number): number {
    91→  return Math.ceil((areaSqm * 1.15) / STANDARD_SHEET_AREA_SQM);
    92→}

    94→/** Fuzzy-match a Site Notebook MaterialTag to inventory items. */
    95→function matchInventoryItems(
    96→  tag: string,
    97→  items: Array<{ id: string; name: string; material: string; stockLevel?: number | string }>
    98→) {
    99→  const tagLower = tag.toLowerCase().trim();
   100→  if (!tagLower) return [];
   101→  // 1. Exact material match
   102→  const exact = items.filter((i) => i.material.toLowerCase() === tagLower);
   103→  if (exact.length > 0) return exact;
   104→  // 2. Substring match on material (either direction)
   105→  const contains = items.filter(
   106→    (i) =>
   107→      i.material.toLowerCase().includes(tagLower) ||
   108→      tagLower.includes(i.material.toLowerCase())
   109→  );
   110→  if (contains.length > 0) return contains;
   111→  // 3. Substring match on item name
   112→  return items.filter((i) => i.name.toLowerCase().includes(tagLower));
   113→}

   281→  // -------- Stock Check feature --------
   282→  // Collects every cutting list entry from every module, groups by material
   283→  // tag, and compares the required sheet count against the total available
   284→  // stock of matching inventory items. Surfaces a "Sync to Inventory"
   285→  // shortcut to create a Stock Request for any shortage.
   286→  async function runStockCheck() {
   287→    setStockCheckLoading(true);
   288→    setStockCheckOpen(true);
   289→    try {
   290→      const inv = await inventoryApi.list();
   291→      const invItems = inv.items;
   292→
   293→      // Aggregate per material tag: total area in m² (length × width × qty, mm→m²)
   294→      const areaByMaterial = new Map<string, number>();
   295→      for (const mod of data.modules) {
   296→        for (const entry of mod.cuttingList) {
   297→          const tag = entry.material || "Standard Board";
   298→          const area = (entry.length * entry.width * entry.qty) / 1_000_000;
   299→          areaByMaterial.set(tag, (areaByMaterial.get(tag) ?? 0) + area);
   300→        }
   301→      }
   302→
   303→      const rows: StockCheckRow[] = Array.from(areaByMaterial.entries()).map(([tag, area]) => {
   304→        const needed = sheetsNeededFor(area);
   305→        const matches = matchInventoryItems(tag, invItems);
   306→        const available = matches.reduce(
   307→          (sum, it) => sum + Number(it.stockLevel ?? 0),
   308→          0
   309→        );
   310→        const shortage = Math.max(0, needed - available);
   311→        return {
   312→          material: tag,
   313→          neededSheets: needed,
   314→          availableSheets: available,
   315→          shortage,
   316→          matchedItemIds: matches.map((m) => m.id),
   317→          matchedItemNames: matches.map((m) => m.name),
   318→        };
   319→      });
   320→
   321→      setStockSummary(rows);
   322→      if (rows.length === 0) {
   323→        toast.info("No cutting list entries to check. Add modules with cutting parts first.");
   324→      } else {
   325→        const shortages = rows.filter((r) => r.shortage > 0).length;
   326→        if (shortages > 0) {
   327→          toast.warning(`Stock check found ${shortages} material(s) with shortages`);
   328→        } else {
   329→          toast.success("Stock check complete — all materials have enough stock");
   330→        }
   331→      }
   332→    } catch (err) {
   333→      console.error("[stock-check] failed:", err);
   334→      toast.error("Failed to run stock check. See console for details.");
   335→    } finally {
   336→      setStockCheckLoading(false);
   337→    }
   338→  }

   340→  // Create a Stock Request covering every shortage that has at least one

   342→  async function syncShortagesToInventory() {
   343→    if (!syncWarehouseId) {
   344→      toast.error("Please select a warehouse to fulfil the request from");
   345→      return;
   346→    }
   347→    const shortages = stockSummary.filter((r) => r.shortage > 0 && r.matchedItemIds.length > 0);
   348→    if (shortages.length === 0) {
   349→      toast.info("No syncable shortages — every shortage is missing a matched inventory item.");
   350→      return;
   351→    }
   352→
   353→    setSyncSubmitting(true);
   354→    try {
   355→      // Use the first matched inventory item per material to keep the request
   356→      // simple. The user can edit the request afterwards if a different item
   357→      // should be used.
   358→      const lines = shortages.map((r) => ({
   359→        itemId: r.matchedItemIds[0],
   360→        quantity: r.shortage,
   361→      }));
   362→
   363→      await stockRequestsApi.create({
   364→        warehouseId: syncWarehouseId,
   365→        jobId: syncJobId || null,
   366→        notes: `Auto-created from Site Notebook (${data.siteName || "untitled"} · ${data.jobNumber || "no job#"})`,
   367→        lines,
   368→      });
   369→
   370→      const unmatchable = stockSummary.filter((r) => r.shortage > 0 && r.matchedItemIds.length === 0);
   371→      if (unmatchable.length > 0) {
   372→        toast.warning(
   373→          `Stock request created for ${shortages.length} material(s). ${unmatchable.length} material(s) had no matching inventory item — please add them manually.`,
   374→          { description: unmatchable.map((r) => r.material).join(", ") }
   375→        );
   376→      } else {
   377→        toast.success(`Stock request created for ${shortages.length} material shortage(s)`);
   378→      }
   379→      setSyncDialogOpen(false);
   380→    } catch (err) {
   381→      console.error("[stock-check] sync failed:", err);
   382→      toast.error(err instanceof Error ? err.message : "Failed to create stock request");
   383→    } finally {
   384→      setSyncSubmitting(false);
   385→    }
   386→  }
