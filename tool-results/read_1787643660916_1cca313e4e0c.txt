   275→  function importJobsCsv(file: File) {
   276→    const reader = new FileReader();
   277→    reader.onload = async () => {
   278→      const text = String(reader.result ?? "");
   279→      const lines = text.split("\n").filter((l) => l.trim());
   280→      if (lines.length < 2) {
   281→        toast.error("CSV must have a header and at least one row");
   282→        return;
   283→      }
   284→      const headers = lines[0].split(",").map((h) => h.trim().toLowerCase().replace(/"/g, ""));
   285→      let imported = 0;
   286→      let failed = 0;
   287→      let customers = customersData?.customers ?? [];
   288→      if (customers.length === 0) {
   289→        try {
   290→          const resp = await customersApi.list();
   291→          customers = resp.customers;
   292→        } catch { /* ignore */ }
   293→      }
   294→      const technicians = usersData?.users ?? [];
   295→      const techMap = new Map(technicians.map((t) => [t.fullName.toLowerCase(), t.id]));
   296→
   297→      for (const line of lines.slice(1)) {
   298→        const cells = line.split(",").map((c) => c.trim().replace(/^"|"$/g, "").replace(/""/g, '"'));
   299→        const get = (key: string) => {
   300→          const idx = headers.indexOf(key);
   301→          return idx >= 0 ? cells[idx] : "";
   302→        };
   303→        const title = get("title");
   304→        const customerName = get("customer name") || get("customer");
   305→        if (!title || !customerName) { failed++; continue; }
   306→        const customer = customers.find((c) => c.name.toLowerCase() === customerName.toLowerCase());
   307→        if (!customer) { failed++; continue; }
   308→        const assignedTo = get("assigned to");
   309→        const assignedToId = assignedTo ? (techMap.get(assignedTo.toLowerCase()) || "") : "";
   310→        const statusVal = (get("status") as JobStatus) || "Pending";
   311→        try {
   312→          // The create endpoint intentionally omits `status` from its
   313→          // payload type for security; we set it via a separate update
   314→          // after the job is created, when the user picked a non-default
   315→          // status in the import template.
   316→          const created = await jobsApi.create({
   317→            title,
   318→            customerId: customer.id,
   319→            assignedToId: assignedToId || undefined,
   320→            priority: (get("priority") as Priority) || "Normal",
   321→            description: get("description") || undefined,
   322→            deliveryDate: get("delivery date") ? new Date(get("delivery date")).toISOString() : undefined,
   323→          });
   324→          if (statusVal && statusVal !== "Pending") {
   325→            try {
   326→              await jobsApi.update(created.job.id, { status: statusVal });
   327→            } catch {
   328→              /* best-effort status patch */
   329→            }
   330→          }
   331→          imported++;
   332→        } catch (e) {
   333→          failed++;
   334→          console.error("Import error:", e);
   335→        }
   336→      }
   337→      toast.success(`Imported ${imported} job(s)${failed > 0 ? `, ${failed} failed` : ""}`);
   338→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   339→    };
   340→    reader.readAsText(file);
   341→  }

   343→  async function duplicateJob(jobId: string) {
   344→    try {
   345→      const original = jobs.find((j) => j.id === jobId);
   346→      if (!original) {
   347→        toast.error("Job not found");
   348→        return;
   349→      }
   350→      const created = await jobsApi.create({
   351→        title: `${original.title} (copy)`,
   352→        customerId: original.customerId,
   353→        priority: original.priority,
   354→        description: original.description ?? undefined,
   355→        deliveryDate: original.deliveryDate ?? undefined,
   356→      });
   357→      toast.success(`Duplicated as ${created.job.orderNumber}`, {
   358→        description: created.job.title,
   359→      });
   360→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   361→      queryClient.invalidateQueries({ queryKey: ["stats"] });
   362→    } catch (e) {
   363→      toast.error(e instanceof Error ? e.message : "Failed to duplicate job");
   364→    }
   365→  }

  1463→/* ============ Calendar View — simple month grid ============ */
  1464→function CalendarView({
  1465→  jobs,
  1466→  onJobClick,
  1467→  onContextMenu,
  1468→}: {
  1469→  jobs: JobOrder[];
  1470→  onJobClick: (id: string) => void;
  1471→  onContextMenu: (id: string, x: number, y: number) => void;
  1472→}) {
  1473→  const [cursor, setCursor] = React.useState(() => {
  1474→    const now = new Date();
  1475→    return new Date(now.getFullYear(), now.getMonth(), 1);
  1476→  });
  1477→
  1478→  const year = cursor.getFullYear();
  1479→  const month = cursor.getMonth();
  1480→  const monthLabel = cursor.toLocaleDateString(undefined, {
  1481→    month: "long",
  1482→    year: "numeric",
  1483→  });
  1484→
  1485→  // Build a 6-week grid starting on Sunday
  1486→  const firstOfMonth = new Date(year, month, 1);
  1487→  const startDay = firstOfMonth.getDay(); // 0..6 (Sun..Sat)
  1488→  const gridStart = new Date(year, month, 1 - startDay);
  1489→  const days: Date[] = [];
  1490→  for (let i = 0; i < 42; i++) {
  1491→    const d = new Date(gridStart);

  1496→  function jobsOnDay(d: Date): JobOrder[] {
  1497→    return jobs.filter((j) => {
  1498→      if (!j.deliveryDate) return false;

  1500→      return (
  1501→        dd.getFullYear() === d.getFullYear() &&
  1502→        dd.getMonth() === d.getMonth() &&
  1503→        dd.getDate() === d.getDate()
  1504→      );
  1505→    });
  1506→  }

  1534→              setCursor(new Date(now.getFullYear(), now.getMonth(), 1));

  1540→            variant="outline"
  1541→            size="icon"
