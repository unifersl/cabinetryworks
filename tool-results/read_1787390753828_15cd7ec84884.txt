    69→export function ElevationCanvas({ data, height = 500, onUpdateModules }: ElevationCanvasProps) {
    70→  const { modules: rawModules, overallDimensions } = data;
    71→  // Guard overall dimensions against NaN/undefined — a common cause of NaN SVG attributes
    72→  const totalW = safeNum(overallDimensions?.width, 3400);
    73→  const totalH = safeNum(overallDimensions?.height, 2150);

    75→  const [zoom, setZoom] = React.useState(1);
    76→  const [selectedId, setSelectedId] = React.useState<string | null>(null);
    77→  const [lockedIds, setLockedIds] = React.useState<Set<string>>(new Set());
    78→  const [contextMenu, setContextMenu] = React.useState<{ id: string; x: number; y: number } | null>(null);
    79→  const [isFullscreen, setIsFullscreen] = React.useState(false);
    80→  const [showHelp, setShowHelp] = React.useState(false);
    81→
    82→  const dragData = React.useRef<{
    83→    id: string; type: "move" | "resize"; edge?: string;
    84→    startPxX: number; startPxY: number;
    85→    origX: number; origY: number; origW: number; origH: number;
    86→  } | null>(null);
    87→  const innerRef = React.useRef<HTMLDivElement>(null);
    88→
    89→  // Normalize legacy modules so drawerHeights / shelfHeights always exist AND
    90→  // all core numeric fields are finite numbers. This is the primary guard
    91→  // against the "Received NaN for attribute" console errors — it migrates any
    92→  // module loaded from localStorage that was saved before these fields existed
    93→  // or that has undefined/NaN width/height/x/y.
    94→  const modules = React.useMemo(
    95→    () => rawModules.map((m) => ensureDrawersShelves(m)),
    96→    [rawModules],
    97→  );
    98→
    99→  const basePxPerMm = Math.min(700 / Math.max(totalW, 1), 350 / Math.max(totalH, 1));
   100→  const pxPerMm = safeNum(basePxPerMm * zoom, 0.2);
   101→  const wallPxW = totalW * pxPerMm;
   102→  const wallPxH = totalH * pxPerMm;
   103→  const PAD = 40;
   104→  const innerW = Math.max(wallPxW + PAD * 2, 700);
   105→  const innerH = Math.max(wallPxH + PAD * 2, 400);
   106→  // NaN-safe mm→px converter: any non-finite input falls back to 0
   107→  const mm = (v: number) => safeNum(v, 0) * pxPerMm;
   108→  const SNAP_THRESHOLD_MM = 50;
   109→
   110→  // --- Accidental close prevention ---
   111→  // When there are modules on the canvas, warn before the tab/window is closed
   112→  // or refreshed so unsaved work isn't lost.

   114→    if (modules.length === 0) return;
   115→    function onBeforeUnload(e: BeforeUnloadEvent) {
   116→      e.preventDefault();
   117→      e.returnValue = "";
   118→    }
   119→    window.addEventListener("beforeunload", onBeforeUnload);
   120→    return () => window.removeEventListener("beforeunload", onBeforeUnload);
   121→  }, [modules.length]);
   122→
   123→  // Esc to exit fullscreen
   124→  React.useEffect(() => {
   125→    if (!isFullscreen) return;

   127→      if (e.key === "Escape") setIsFullscreen(false);
   128→    }
   129→    window.addEventListener("keydown", onKey);
   130→    return () => window.removeEventListener("keydown", onKey);
   131→  }, [isFullscreen]);

   133→  // --- Snap helper: find nearby edges to snap to ---
   134→  function getSnappedPos(modId: string, newX: number, newY: number, newW: number, newH: number): { x: number; y: number } {
   135→    let snapX = newX, snapY = newY;
   136→    const myEdges = {
   137→      left: newX, right: newX + newW,
   138→      top: newY, bottom: newY + newH,
   139→      centerX: newX + newW / 2, centerY: newY + newH / 2,
   140→    };
   141→    for (const other of modules) {
   142→      if (other.id === modId) continue;
   143→      const oEdges = {
   144→        left: other.x, right: other.x + other.width,
   145→        top: other.y, bottom: other.y + other.height,
   146→        centerX: other.x + other.width / 2, centerY: other.y + other.height / 2,
   147→      };
   148→      for (const [myKey, myVal] of Object.entries(myEdges)) {
   149→        if (!myKey.includes("x") && myKey !== "left" && myKey !== "right" && myKey !== "centerX") continue;
   150→        for (const [oKey, oVal] of Object.entries(oEdges)) {
   151→          if (!oKey.includes("x") && oKey !== "left" && oKey !== "right" && oKey !== "centerX") continue;
   152→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
   153→            if (myKey === "left" || myKey === "right" || myKey === "centerX") {
   154→              const diff = oVal - myVal;
   155→              snapX = newX + diff;
   156→            }
   157→          }
   158→        }
   159→      }
   160→      for (const [myKey, myVal] of Object.entries(myEdges)) {
   161→        if (myKey !== "top" && myKey !== "bottom" && myKey !== "centerY") continue;
   162→        for (const [oKey, oVal] of Object.entries(oEdges)) {
   163→          if (oKey !== "top" && oKey !== "bottom" && oKey !== "centerY") continue;
   164→          if (Math.abs(myVal - oVal) < SNAP_THRESHOLD_MM) {
   165→            if (myKey === "top" || myKey === "bottom" || myKey === "centerY") {
   166→              const diff = oVal - myVal;
   167→              snapY = newY + diff;
   168→            }
   169→          }
   170→        }
   171→      }
   172→    }
   173→    if (Math.abs(snapX) < SNAP_THRESHOLD_MM) snapX = 0;
   174→    if (Math.abs(snapY) < SNAP_THRESHOLD_MM) snapY = 0;
   175→    if (Math.abs(snapX + newW - totalW) < SNAP_THRESHOLD_MM) snapX = totalW - newW;
   176→    if (Math.abs(snapY + newH - totalH) < SNAP_THRESHOLD_MM) snapY = totalH - newH;
   177→    return { x: snapX, y: snapY };
   178→  }

   180→  function handleDrop(e: React.DragEvent) {
   181→    e.preventDefault();
   182→    const type = e.dataTransfer.getData("text/cabinet-type");
   183→    if (!type || !onUpdateModules) return;
   184→    const item = PALETTE.find((p) => p.type === type);
   185→    if (!item) return;
   186→    const inner = innerRef.current;
   187→    if (!inner) return;
   188→    const rect = inner.getBoundingClientRect();
   189→    const mmX = (e.clientX - rect.left - PAD) / pxPerMm;
   190→    const mmY = (e.clientY - rect.top - PAD) / pxPerMm;
   191→    const newCode = String(modules.length + 1).padStart(2, "0");
   192→    const newMod = createEmptyModule(newCode);
   193→    newMod.type = type as any;
   194→    newMod.width = item.w;
   195→    newMod.height = item.h;
   196→    newMod.x = Math.round(Math.max(0, Math.min(totalW - item.w, mmX - item.w / 2)));
   197→    newMod.y = Math.round(Math.max(0, Math.min(totalH - item.h, mmY - item.h / 2)));
   198→    onUpdateModules([...modules, newMod]);
   199→    setSelectedId(newMod.id);
   200→  }

   202→  function onModuleDown(e: React.PointerEvent, mod: CabinetModule) {
   203→    if (!onUpdateModules) return;
   204→    e.stopPropagation();
   205→    setSelectedId(mod.id);
   206→    if (lockedIds.has(mod.id)) return;
   207→    dragData.current = {
   208→      id: mod.id, type: "move",
   209→      startPxX: e.clientX, startPxY: e.clientY,
   210→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   211→    };
   212→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   213→  }

   215→  function onResizeDown(e: React.PointerEvent, mod: CabinetModule, edge: string) {
   216→    e.stopPropagation();
   217→    setSelectedId(mod.id);
   218→    if (lockedIds.has(mod.id)) return;
   219→    dragData.current = {
   220→      id: mod.id, type: "resize", edge,
   221→      startPxX: e.clientX, startPxY: e.clientY,
   222→      origX: mod.x, origY: mod.y, origW: mod.width, origH: mod.height,
   223→    };
   224→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   225→  }

   227→  function onCanvasMove(e: React.PointerEvent) {
   228→    const dd = dragData.current;
   229→    if (!dd || !onUpdateModules) return;
   230→    const dxMm = (e.clientX - dd.startPxX) / pxPerMm;
   231→    const dyMm = (e.clientY - dd.startPxY) / pxPerMm;
   232→    if (dd.type === "move") {
   233→      let newX = Math.max(0, Math.min(totalW - 50, Math.round(dd.origX + dxMm)));
   234→      let newY = Math.max(0, Math.min(totalH - 50, Math.round(dd.origY + dyMm)));
   235→      const snapped = getSnappedPos(dd.id, newX, newY, dd.origW, dd.origH);
   236→      newX = snapped.x; newY = snapped.y;
   237→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, x: newX, y: newY } : m));
   238→    } else if (dd.type === "resize") {
   239→      const min = 100;
   240→      let nw = dd.origW, nh = dd.origH, nx = dd.origX, ny = dd.origY;
   241→      if (dd.edge === "right") nw = Math.max(min, Math.round(dd.origW + dxMm));
   242→      if (dd.edge === "left") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); }
   243→      if (dd.edge === "bottom") nh = Math.max(min, Math.round(dd.origH + dyMm));
   244→      if (dd.edge === "top") { nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   245→      if (dd.edge === "br") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   246→      if (dd.edge === "bl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH + dyMm)); }
   247→      if (dd.edge === "tr") { nw = Math.max(min, Math.round(dd.origW + dxMm)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   248→      if (dd.edge === "tl") { nw = Math.max(min, Math.round(dd.origW - dxMm)); nx = Math.round(dd.origX + (dd.origW - nw)); nh = Math.max(min, Math.round(dd.origH - dyMm)); ny = Math.round(dd.origY + (dd.origH - nh)); }
   249→      onUpdateModules(modules.map((m) => m.id === dd.id ? { ...m, width: nw, height: nh, x: nx, y: ny } : m));
   250→    }
   251→  }

   255→  function deleteModule(id: string) {
   256→    if (!onUpdateModules) return;
   257→    onUpdateModules(modules.filter((m) => m.id !== id).map((m, i) => ({ ...m, code: String(i + 1).padStart(2, "0") })));
   258→    setSelectedId(null);
   259→    setLockedIds((s) => { const n = new Set(s); n.delete(id); return n; });
   260→  }

   262→  function duplicateModule(id: string) {
   263→    if (!onUpdateModules) return;
   264→    const mod = modules.find((m) => m.id === id);
   265→    if (!mod) return;
   266→    const newCode = String(modules.length + 1).padStart(2, "0");
   267→    const copy: CabinetModule = {
   268→      ...mod,
   269→      id: crypto.randomUUID(),
   270→      code: newCode,
   271→      x: mod.x + 50,
   272→      y: mod.y + 50,
   273→      cuttingList: mod.cuttingList.map(c => ({ ...c, id: crypto.randomUUID() })),
   274→      drawerHeights: [...(mod.drawerHeights ?? [])],
   275→      shelfHeights: [...(mod.shelfHeights ?? [])],
   276→    };
   277→    onUpdateModules([...modules, copy]);
   278→    setSelectedId(copy.id);
   279→    toast.success(`Duplicated Module ${mod.code} → ${newCode}`);
   280→  }

   292→  function setDrawerCount(id: string, count: number) {
   293→    const c = Math.max(1, Math.min(10, Math.floor(count)));
   294→    const mod = modules.find((m) => m.id === id);
   295→    if (!mod) return;
   296→    patchModule(id, { drawerHeights: distributeEvenly(c, mod.height, 0) });

   299→  function setDrawerHeight(id: string, idx: number, h: number) {
   300→    const mod = modules.find((m) => m.id === id);
   301→    if (!mod) return;
   302→    const next = [...(mod.drawerHeights ?? [])];
   303→    next[idx] = Math.max(50, Math.round(h));
   304→    patchModule(id, { drawerHeights: next });
   305→  }

   307→  function setShelfCount(id: string, count: number) {
   308→    const c = Math.max(0, Math.min(8, Math.floor(count)));
   309→    const mod = modules.find((m) => m.id === id);
   310→    if (!mod) return;
   311→    const positions = Array.from({ length: c }, (_, i) =>
   312→      Math.round((mod.height * (i + 1)) / (c + 1)),
   313→    );
   314→    patchModule(id, { shelfHeights: positions });
   315→  }

   317→  function setShelfHeight(id: string, idx: number, hFromBottom: number) {
   318→    const mod = modules.find((m) => m.id === id);
   319→    if (!mod) return;
   320→    const next = [...(mod.shelfHeights ?? [])];
   321→    next[idx] = Math.max(20, Math.min(mod.height - 20, Math.round(hFromBottom)));
   322→    patchModule(id, { shelfHeights: next });
   323→  }

   340→  function renderShape(mod: CabinetModule, w: number, h: number) {
   341→    const show = w > 25 && h > 25;
   342→    const drawers = Array.isArray(mod.drawerHeights) ? mod.drawerHeights : [];
   343→    const shelves = Array.isArray(mod.shelfHeights) ? mod.shelfHeights : [];
   344→    const t = mod.type;
   345→
   346→    // Determine what to show based on type
   347→    const isAppliance = APPLIANCE_TYPES.has(t);
   348→    const isOpenShelf = OPEN_SHELVING_TYPES.has(t);
   349→    const isTallUnit = TALL_UNIT_TYPES.has(t);
   350→
   351→    // Shelves only for standard cabinets and open shelving
   352→    const showShelves = !isAppliance && !isTallUnit && !isOpenShelf
   353→      ? (mod.openingStyle !== "Drawers" && mod.openingStyle !== "Up-Lift Flap")
   354→      : isOpenShelf;
   355→
   356→    // Door lines only for standard cabinets (not appliances, not open shelving)
   357→    const showDoors = !isAppliance && !isOpenShelf && !isTallUnit;
   358→
   359→    return (
   360→      <svg width={w} height={h} style={{ position: "absolute", top: 0, left: 0, pointerEvents: "none", overflow: "visible" }}>
   361→        {/* Door split lines — ONLY for standard cabinets */}
   362→        {showDoors && mod.openingStyle === "Single Door" && show && <line x1={w/2} y1={3} x2={w/2} y2={h-3} stroke="#94a3b8" strokeWidth="1" strokeDasharray="3 2" />}
   363→        {showDoors && mod.openingStyle === "Double Door" && show && (<><line x1={w*0.33} y1={3} x2={w*0.33} y2={h-3} stroke="#94a3b8" strokeWidth="1" strokeDasharray="3 2" /><line x1={w*0.66} y1={3} x2={w*0.66} y2={h-3} stroke="#94a3b8" strokeWidth="1" strokeDasharray="3 2" /></>)}
   364→
   365→        {/* Drawers — ONLY for standard cabinets with Drawers opening style */}
   366→        {showDoors && mod.openingStyle === "Drawers" && show && (() => {
   367→          let cumY = 0;
   368→          return drawers.map((dh, i) => {
   369→            const drawerPxH = Math.max(2, safeNum(dh, 0) * pxPerMm);
   370→            const top = cumY;
   371→            cumY += drawerPxH;
