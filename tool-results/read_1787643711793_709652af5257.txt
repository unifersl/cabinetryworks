   186→/* ---------- Helpers ---------- */
   187→function num(v: unknown): number {
   188→  const n = typeof v === "string" ? parseFloat(v) : Number(v);
   189→  return Number.isFinite(n) ? n : 0;
   190→}
   191→
   192→function fmtLKR(v: unknown): string {
   193→  const n = num(v);
   194→  if (!n) return "—";
   195→  return `${n.toLocaleString()} LKR`;
   196→}
   197→
   198→function fmtDate(s?: string | null): string {
   199→  if (!s) return "—";
   200→  try {
   201→    return new Date(s).toLocaleDateString(undefined, {
   202→      year: "numeric",
   203→      month: "short",
   204→      day: "numeric",
   205→    });
   206→  } catch {
   207→    return "—";
   208→  }
   209→}
   210→
   211→function fmtDateTime(s?: string | null): string {
   212→  if (!s) return "—";
   213→  try {
   214→    return new Date(s).toLocaleString(undefined, {
   215→      month: "short",
   216→      day: "numeric",
   217→      hour: "2-digit",
   218→      minute: "2-digit",
   219→    });
   220→  } catch {
   221→    return "—";
   222→  }
   223→}
   224→
   225→function toInputDate(s?: string | null): string {
   226→  if (!s) return "";
   227→  try {
   228→    const d = new Date(s);
   229→    if (Number.isNaN(d.getTime())) return "";
   230→    return d.toISOString().slice(0, 10);
   231→  } catch {
   232→    return "";
   233→  }
   234→}
   235→
   236→function toInputDateTimeLocal(s?: string | null): string {
   237→  if (!s) return "";
   238→  try {
   239→    const d = new Date(s);
   240→    if (Number.isNaN(d.getTime())) return "";
   241→    // local time
   242→    const offsetMs = d.getTimezoneOffset() * 60000;
   243→    return new Date(d.getTime() - offsetMs).toISOString().slice(0, 16);
   244→  } catch {
   245→    return "";
   246→  }
   247→}

   952→function TimeLogDialog({
   953→  jobId,
   954→  log,
   955→  users,
   956→  open,
   957→  onOpenChange,
   958→}: {
   959→  jobId: string;
   960→  log: JobTimeLog | null;
   961→  users: { id: string; fullName: string; role: string }[];
   962→  open: boolean;
   963→  onOpenChange: (o: boolean) => void;
   964→}) {
   965→  const queryClient = useQueryClient();
   966→  const [workerName, setWorkerName] = React.useState("");
   967→  const [userId, setUserId] = React.useState("");
   968→  const [workDate, setWorkDate] = React.useState("");
   969→  const [workType, setWorkType] = React.useState("onsite");
   970→  const [clockIn, setClockIn] = React.useState("");
   971→  const [clockOut, setClockOut] = React.useState("");
   972→  const [hoursWorked, setHoursWorked] = React.useState("");
   973→  const [hourlyRate, setHourlyRate] = React.useState("");
   974→  const [notes, setNotes] = React.useState("");
   975→  const [useClock, setUseClock] = React.useState(true);
   976→
   977→  React.useEffect(() => {
   978→    if (log) {
   979→      setWorkerName(log.workerName);
   980→      setUserId(log.userId ?? "");
   981→      setWorkDate(toInputDate(log.workDate));
   982→      setWorkType(log.workType);
   983→      setClockIn(toInputDateTimeLocal(log.clockIn));
   984→      setClockOut(toInputDateTimeLocal(log.clockOut));
   985→      setHoursWorked(log.clockIn && log.clockOut ? "" : String(log.hoursWorked));
   986→      setHourlyRate(num(log.hourlyRate) > 0 ? String(log.hourlyRate) : "");
   987→      setNotes(log.notes ?? "");
   988→      setUseClock(!!log.clockIn && !!log.clockOut);
   989→    } else {
   990→      setWorkerName("");
   991→      setUserId("");
   992→      setWorkDate(new Date().toISOString().slice(0, 10));
   993→      setWorkType("onsite");
   994→      setClockIn("");
   995→      setClockOut("");
   996→      setHoursWorked("");
   997→      setHourlyRate("");
   998→      setNotes("");
   999→      setUseClock(true);
  1000→    }
  1001→  }, [log, open]);
  1002→
  1003→  // Auto-set worker name from selected user
  1004→  React.useEffect(() => {
  1005→    if (userId) {
  1006→      const u = users.find((x) => x.id === userId);
  1007→      if (u) setWorkerName(u.fullName);
  1008→    }
  1009→  }, [userId, users]);
  1010→
  1011→  const save = useMutation({
  1012→    mutationFn: async () => {
  1013→      const payload: Record<string, unknown> = {
  1014→        workerName: workerName.trim(),
  1015→        workType,
  1016→        workDate: workDate || new Date().toISOString(),
  1017→        notes: notes || undefined,
  1018→      };
  1019→      if (userId) payload.userId = userId;
  1020→      if (useClock) {
  1021→        if (clockIn) payload.clockIn = new Date(clockIn).toISOString();
  1022→        if (clockOut) payload.clockOut = new Date(clockOut).toISOString();
  1023→      } else if (hoursWorked) {
  1024→        payload.hoursWorked = Number(hoursWorked);
  1025→      }
  1026→      const rate = Number(hourlyRate);
  1027→      if (Number.isFinite(rate) && rate > 0) payload.hourlyRate = rate;
  1028→      if (log) {
  1029→        return jobTimeLogsApi.update(log.id, payload);
  1030→      }
  1031→      return jobTimeLogsApi.create({ jobId, ...payload } as Parameters<typeof jobTimeLogsApi.create>[0]);
  1032→    },
  1033→    onSuccess: () => {
  1034→      toast.success(log ? "Time log updated" : "Time log added");
  1035→      queryClient.invalidateQueries({ queryKey: ["job-time-logs", jobId] });
  1036→      queryClient.invalidateQueries({ queryKey: ["job-insights", jobId] });
  1037→      onOpenChange(false);
  1038→    },
  1039→    onError: (e: Error) => toast.error(e.message),
  1040→  });
  1041→
  1042→  return (
  1043→    <Dialog open={open} onOpenChange={onOpenChange}>
  1044→      <DialogContent showCloseButton={false} className="sm:max-w-[480px]">
  1045→        <DialogHeader>
  1046→          <div className="flex items-start justify-between">
  1047→            <div>
  1048→              <DialogTitle>{log ? "Edit Time Log" : "Add Time Log"}</DialogTitle>
  1049→              <DialogDescription>Track worker hours for this job.</DialogDescription>
  1050→            </div>
  1051→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => onOpenChange(false)}>
  1052→              <X className="h-4 w-4" />
  1053→            </Button>
  1054→          </div>
  1055→        </DialogHeader>
  1056→        <form
  1057→          onSubmit={(e) => {
  1058→            e.preventDefault();
  1059→            if (!workerName.trim()) {
  1060→              toast.error("Worker name is required");
  1061→              return;
  1062→            }
  1063→            save.mutate();
  1064→          }}
  1065→          className="space-y-3"
  1066→        >
  1067→          <div className="grid grid-cols-2 gap-3">
  1068→            <div className="space-y-1.5">
  1069→              <Label className="text-xs">Worker (system user)</Label>
  1070→              <Select value={userId} onValueChange={(v) => setUserId(v === "__none" ? "" : v)}>
  1071→                <SelectTrigger className="h-9"><SelectValue placeholder="—" /></SelectTrigger>
  1072→                <SelectContent>
  1073→                  <SelectItem value="__none">— Free text —</SelectItem>
  1074→                  {users.map((u) => (
  1075→                    <SelectItem key={u.id} value={u.id}>{u.fullName} ({u.role})</SelectItem>
  1076→                  ))}
  1077→                </SelectContent>
  1078→              </Select>
  1079→            </div>
  1080→            <div className="space-y-1.5">
  1081→              <Label className="text-xs">Worker Name *</Label>
  1082→              <Input
  1083→                className="h-9"
  1084→                value={workerName}
  1085→                onChange={(e) => setWorkerName(e.target.value)}
  1086→                placeholder="e.g. John Smith"
  1087→                required
  1088→              />
  1089→            </div>
  1090→            <div className="space-y-1.5">
  1091→              <Label className="text-xs">Date</Label>
  1092→              <Input
  1093→                type="date"
  1094→                className="h-9"
  1095→                value={workDate}
  1096→                onChange={(e) => setWorkDate(e.target.value)}
  1097→                required
