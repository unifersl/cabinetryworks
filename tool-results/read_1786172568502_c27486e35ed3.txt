    93→  function updateData(patch: Partial<SiteJobData>) {
    94→    setData((d) => ({ ...d, ...patch }));
    95→  }

    97→  function addModule() {
    98→    const code = String(data.modules.length + 1).padStart(2, "0");
    99→    const newMod = createEmptyModule(code);
   100→    setData((d) => ({ ...d, modules: [...d.modules, newMod] }));
   101→    setExpandedModules((s) => new Set([...s, newMod.id]));
   102→    toast.success(`Module ${code} added`);
   103→  }
   104→
   105→  function updateModule(id: string, patch: Partial<CabinetModule>) {
   106→    setData((d) => ({
   107→      ...d,
   108→      modules: d.modules.map((m) => (m.id === id ? { ...m, ...patch } : m)),
   109→    }));
   110→  }
   111→
   112→  function deleteModule(id: string) {
   113→    setData((d) => ({
   114→      ...d,
   115→      modules: d.modules
   116→        .filter((m) => m.id !== id)
   117→        .map((m, i) => ({ ...m, code: String(i + 1).padStart(2, "0") })),
   118→    }));
   119→    toast.success("Module removed");
   120→  }
   121→
   122→  function duplicateModule(id: string) {
   123→    const mod = data.modules.find((m) => m.id === id);
   124→    if (!mod) return;
   125→    const idx = data.modules.indexOf(mod);
   126→    const newMod: CabinetModule = {
   127→      ...mod,
   128→      id: crypto.randomUUID(),
   129→      cuttingList: mod.cuttingList.map((c) => ({ ...c, id: crypto.randomUUID() })),
   130→    };
   131→    const newModules = [...data.modules];
   132→    newModules.splice(idx + 1, 0, newMod);
   133→    // Recode
   134→    newModules.forEach((m, i) => (m.code = String(i + 1).padStart(2, "0")));
   135→    setData((d) => ({ ...d, modules: newModules }));
   136→    toast.success(`Module ${mod.code} duplicated`);
   137→  }
   138→
   139→  function toggleExpand(id: string) {
   140→    setExpandedModules((s) => {
   141→      const next = new Set(s);
   142→      if (next.has(id)) next.delete(id);
   143→      else next.add(id);
   144→      return next;
   145→    });
   146→  }

   148→  function addCuttingEntry(moduleId: string) {
   149→    const mod = data.modules.find((m) => m.id === moduleId);
   150→    if (!mod) return;
   151→    const newEntry = {
   152→      id: crypto.randomUUID(),
   153→      length: 580,
   154→      width: 560,
   155→      qty: 2,
   156→      material: "Standard Board" as MaterialTag,
   157→      note: "Side Panel",
   158→    };
   159→    updateModule(moduleId, { cuttingList: [...mod.cuttingList, newEntry] });
   160→  }

   162→  function updateCuttingEntry(moduleId: string, entryId: string, patch: Partial<{ length: number; width: number; qty: number; material: MaterialTag; note: string }>) {
   163→    const mod = data.modules.find((m) => m.id === moduleId);
   164→    if (!mod) return;
   165→    updateModule(moduleId, {
   166→      cuttingList: mod.cuttingList.map((e) => (e.id === entryId ? { ...e, ...patch } : e)),
   167→    });
   168→  }

   170→  function deleteCuttingEntry(moduleId: string, entryId: string) {
   171→    const mod = data.modules.find((m) => m.id === moduleId);
   172→    if (!mod) return;
   173→    updateModule(moduleId, { cuttingList: mod.cuttingList.filter((e) => e.id !== entryId) });
   174→  }

   176→  function exportJSON() {
   177→    const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
   178→    const url = URL.createObjectURL(blob);
   179→    const a = document.createElement("a");
   180→    a.href = url;
   181→    a.download = `site-notebook-${data.siteName || "untitled"}-${data.jobNumber || Date.now()}.json`;
   182→    a.click();
   183→    URL.revokeObjectURL(url);
   184→    toast.success("Site data exported as JSON");
   185→  }

   187→  function importJSON(e: React.ChangeEvent<HTMLInputElement>) {
   188→    const file = e.target.files?.[0];
   189→    if (!file) return;
   190→    const reader = new FileReader();
   191→    reader.onload = (ev) => {
   192→      try {
   193→        const parsed = JSON.parse(ev.target?.result as string) as SiteJobData;
   194→        if (!parsed.modules) throw new Error("Invalid format");
   195→        setData(parsed);
   196→        toast.success("Site data imported");
   197→      } catch {
   198→        toast.error("Invalid JSON file");
   199→      }
   200→    };
   201→    reader.readAsText(file);
   202→    e.target.value = "";
   203→  }
   204→
   205→  function handlePhotoUpload(e: React.ChangeEvent<HTMLInputElement>) {
   206→    const files = e.target.files;
   207→    if (!files) return;
   208→    Array.from(files).forEach((file) => {
   209→      const reader = new FileReader();
   210→      reader.onload = (ev) => {
   211→        setPhotos((p) => [...p, ev.target?.result as string]);
   212→      };
   213→      reader.readAsDataURL(file);
   214→    });
   215→    e.target.value = "";
   216→  }
