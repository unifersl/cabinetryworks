   118→export function JobOrdersView({
   119→  focusJobId,
   120→  onFocusConsumed,
   121→}: {
   122→  focusJobId?: string | null;
   123→  onFocusConsumed?: () => void;
   124→} = {}) {
   125→  const { user } = useAuth();

   127→  const queryClient = useQueryClient();
   128→  const [createOpen, setCreateOpen] = React.useState(false);
   129→  const [search, setSearch] = React.useState("");
   130→  const [statusFilter, setStatusFilter] = React.useState<"all" | JobStatus>(
   131→    "all"
   132→  );
   133→  const [selectedJobId, setSelectedJobId] = React.useState<string | null>(null);
   134→  const [detailOpen, setDetailOpen] = React.useState(false);
   135→
   136→  // Handle focus from global search
   137→  React.useEffect(() => {
   138→    if (focusJobId) {
   139→      setSelectedJobId(focusJobId);
   140→      setDetailOpen(true);
   141→      onFocusConsumed?.();
   142→    }
   143→  }, [focusJobId, onFocusConsumed]);
   144→
   145→  const [showArchived, setShowArchived] = React.useState(false);
   146→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
   147→
   148→  const { data: jobsData, isLoading } = useQuery({
   149→    queryKey: ["jobs", statusFilter, showArchived],
   150→    queryFn: () =>
   151→      jobsApi.list(
   152→        showArchived
   153→          ? { archived: true }
   154→          : statusFilter === "all"
   155→            ? undefined
   156→            : { status: statusFilter }
   157→      ),
   158→  });
   159→
   160→  const { data: customersData } = useQuery({
   161→    queryKey: ["customers"],

   163→  });

   165→    queryKey: ["users"],

   182→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   183→      queryClient.invalidateQueries({ queryKey: ["stats"] });
   184→      toast.success("Job status updated");
   185→    },
   186→    onError: (e: Error) => toast.error(e.message),
   187→  });

   189→  // ---- CSV Template / Import / Export ----
   190→  function downloadJobTemplate() {
   191→    const headers = ["Title", "Customer Name", "Status", "Priority", "Assigned To", "Description", "Delivery Date"];
   192→    const sampleRows = [

   194→      ["Wardrobe Installation", "Jane Doe", "Design", "High", "technician", "3-door wardrobe", "2026-09-20"],

   196→    ];
   197→    const csv = [headers.join(","), ...sampleRows.map((r) => r.map((c) => `"${c}"`).join(","))].join("\n");

   208→  function exportJobsCsv() {
   209→    const headers = ["Order Number", "Title", "Customer", "Status", "Priority", "Assigned To", "Delivery Date", "Created"];
   210→    const rows = (jobsData?.jobs ?? []).map((j) => [
   211→      j.orderNumber,
   212→      j.title,
   213→      j.customer?.name ?? "",
   214→      j.status,
   215→      j.priority,
   216→      j.assignedTo?.fullName ?? "",
   217→      j.deliveryDate ? new Date(j.deliveryDate).toLocaleDateString("en-GB") : "",
   218→      new Date(j.createdAt).toLocaleDateString("en-GB"),
   219→    ]);
   220→    const csv = [headers.join(","), ...rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(","))].join("\n");
   221→    const blob = new Blob([csv], { type: "text/csv" });
   222→    const url = URL.createObjectURL(blob);
   223→    const a = document.createElement("a");
   224→    a.href = url;
   225→    a.download = `job-orders-${new Date().toISOString().slice(0, 10)}.csv`;
   226→    a.click();
   227→    URL.revokeObjectURL(url);
   228→    toast.success("CSV exported");
   229→  }
   230→
   231→  function importJobsCsv(file: File) {
   232→    const reader = new FileReader();
   233→    reader.onload = async () => {
   234→      const text = String(reader.result ?? "");
   235→      const lines = text.split("\n").filter((l) => l.trim());
   236→      if (lines.length < 2) {
   237→        toast.error("CSV must have a header and at least one row");
   238→        return;
   239→      }
   240→      const headers = lines[0].split(",").map((h) => h.trim().toLowerCase().replace(/"/g, ""));
   241→      let imported = 0;
   242→      let failed = 0;
   243→      let customers = customersData?.customers ?? [];
   244→      if (customers.length === 0) {
   245→        try {
   246→          const resp = await customersApi.list();
   247→          customers = resp.customers;
   248→        } catch { /* ignore */ }
   249→      }
   250→      const technicians = usersData?.users ?? [];
   251→      const techMap = new Map(technicians.map((t) => [t.fullName.toLowerCase(), t.id]));
   252→
   253→      for (const line of lines.slice(1)) {
   254→        const cells = line.split(",").map((c) => c.trim().replace(/^"|"$/g, "").replace(/""/g, '"'));
   255→        const get = (key: string) => {
   256→          const idx = headers.indexOf(key);
   257→          return idx >= 0 ? cells[idx] : "";
   258→        };
   259→        const title = get("title");
   260→        const customerName = get("customer name") || get("customer");
   261→        if (!title || !customerName) { failed++; continue; }
   262→        const customer = customers.find((c) => c.name.toLowerCase() === customerName.toLowerCase());
   263→        if (!customer) { failed++; continue; }
   264→        const assignedTo = get("assigned to");
   265→        const assignedToId = assignedTo ? (techMap.get(assignedTo.toLowerCase()) || "") : "";
   266→        try {
   267→          await jobsApi.create({
   268→            title,
   269→            customerId: customer.id,
   270→            status: (get("status") as JobStatus) || "Pending",
   271→            priority: (get("priority") as Priority) || "Normal",
   272→            assignedToId: assignedToId || undefined,
   273→            description: get("description") || undefined,
   274→            deliveryDate: get("delivery date") ? new Date(get("delivery date")).toISOString() : undefined,
   275→          });
   276→          imported++;
   277→        } catch (e) {
   278→          failed++;
   279→          console.error("Import error:", e);
   280→        }
   281→      }
   282→      toast.success(`Imported ${imported} job(s)${failed > 0 ? `, ${failed} failed` : ""}`);
   283→      queryClient.invalidateQueries({ queryKey: ["jobs"] });
   284→    };
   285→    reader.readAsText(file);
   286→  }
   287→
   288→  return (
   289→    <div className="space-y-4">
   290→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   291→        <div>
   292→          <h1 className="text-lg font-bold tracking-tight">Job Orders</h1>
   293→          <p className="text-sm text-muted-foreground">
   294→            Track cabinetry production jobs through the full pipeline.
   295→          </p>
   296→        </div>
   297→        <div className="flex flex-wrap gap-2">
   298→          <input
   299→            type="file"
   300→            accept=".csv"
   301→            id="job-import"
   302→            className="hidden"
   303→            onChange={(e) => {
   304→              const f = e.target.files?.[0];
   305→              if (f) importJobsCsv(f);
   306→              e.target.value = "";
   307→            }}
   308→          />
   309→          <Button variant="outline" size="sm" onClick={() => document.getElementById("job-import")?.click()}>
   310→            <Upload className="mr-1.5 h-4 w-4" /> Import
   311→          </Button>
   312→          <Button variant="outline" size="sm" onClick={exportJobsCsv}>
   313→            <Download className="mr-1.5 h-4 w-4" /> Export
   314→          </Button>
   315→          <Button variant="ghost" size="sm" onClick={downloadJobTemplate} title="Download CSV template with sample data">
   316→            <FileSpreadsheet className="mr-1.5 h-4 w-4" /> Template
   317→          </Button>
   318→          <Button onClick={() => setCreateOpen(true)}>
   319→            <Plus className="mr-2 h-4 w-4" />
   320→            New Job Order
   321→          </Button>
   322→        </div>
   323→      </div>
   324→
   325→      {/* Bulk action bar */}
   326→      {selectedIds.size > 0 && (
   327→        <div className="flex flex-wrap items-center gap-2 rounded-lg border border-primary/30 bg-primary/5 p-3">
   328→          <span className="text-sm font-medium">
   329→            {selectedIds.size} selected
   330→          </span>
   331→          <div className="ml-auto flex flex-wrap gap-2">
   332→            <Button
   333→              variant="outline"
   334→              size="sm"
   335→              onClick={async () => {
   336→                try {
   337→                  await bulkApi.updateJobs([...selectedIds], showArchived ? "restore" : "archive");
   338→                  toast.success(`${selectedIds.size} job(s) ${showArchived ? "restored" : "archived"}`);
   339→                  setSelectedIds(new Set());
   340→                  queryClient.invalidateQueries({ queryKey: ["jobs"] });
   341→                } catch (e) {
   342→                  toast.error(e instanceof Error ? e.message : "Bulk action failed");
   343→                }
   344→              }}
   345→            >
   346→              <Archive className="mr-1.5 h-3.5 w-3.5" />
   347→              {showArchived ? "Restore" : "Archive"}
   348→            </Button>
   349→            <Select
   350→              onValueChange={async (status) => {
   351→                try {
   352→                  await bulkApi.updateJobs([...selectedIds], "status", status);
   353→                  toast.success(`${selectedIds.size} job(s) → ${status}`);
   354→                  setSelectedIds(new Set());
   355→                  queryClient.invalidateQueries({ queryKey: ["jobs"] });
   356→                } catch (e) {
   357→                  toast.error(e instanceof Error ? e.message : "Bulk action failed");
   358→                }
   359→              }}
   360→            >
   361→              <SelectTrigger className="h-8 w-[140px]">
   362→                <Filter className="mr-1.5 h-3.5 w-3.5" />
   363→                <span className="text-xs">Set status…</span>
   364→              </SelectTrigger>
   365→              <SelectContent>
   366→                {STATUSES.map((s) => (
   367→                  <SelectItem key={s} value={s}>
   368→                    {s}
   369→                  </SelectItem>
   370→                ))}
   371→              </SelectContent>
   372→            </Select>
   373→            {isAdmin && (
   374→              <AlertDialog>
   375→                <AlertDialogTrigger asChild>
   376→                  <Button variant="destructive" size="sm">
   377→                    <Trash2 className="mr-1.5 h-3.5 w-3.5" />
   378→                    Delete Selected
   379→                  </Button>
   380→                </AlertDialogTrigger>
   381→                <AlertDialogContent>
   382→                  <AlertDialogHeader>
   383→                    <AlertDialogTitle>Delete {selectedIds.size} job(s) permanently?</AlertDialogTitle>
   384→                    <AlertDialogDescription>
   385→                      This will permanently delete all selected jobs and ALL their related data (time logs, transport, F&B, expenses, tools, stock movements, measurements, cutting lists, quotes). This action cannot be undone.
   386→                    </AlertDialogDescription>
   387→                  </AlertDialogHeader>
   388→                  <AlertDialogFooter>
   389→                    <AlertDialogCancel>Cancel</AlertDialogCancel>
   390→                    <AlertDialogAction
   391→                      className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   392→                      onClick={async () => {
   393→                        let ok = 0;
   394→                        let fail = 0;
   395→                        for (const id of [...selectedIds]) {
   396→                          try {
   397→                            await jobsApi.remove(id);
   398→                            ok++;
   399→                          } catch {
   400→                            fail++;
   401→                          }
   402→                        }
   403→                        toast.success(`${ok} job(s) deleted${fail > 0 ? `, ${fail} failed` : ""}`);
   404→                        setSelectedIds(new Set());
   405→                        queryClient.invalidateQueries({ queryKey: ["jobs"] });
   406→                      }}
   407→                    >
   408→                      <Trash2 className="mr-1.5 h-4 w-4" />
   409→                      Delete {selectedIds.size} Job(s)
   410→                    </AlertDialogAction>
   411→                  </AlertDialogFooter>
   412→                </AlertDialogContent>
   413→              </AlertDialog>
   414→            )}
   415→            <Button
   416→              variant="ghost"
   417→              size="sm"
   418→              onClick={() => setSelectedIds(new Set())}
   419→            >
   420→              Clear
   421→            </Button>
   422→          </div>
   423→        </div>
   424→      )}
   425→
   426→      <Card>
   427→        <CardHeader className="pb-2">
   428→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   429→            <div>
   430→              <CardTitle className="text-base">All Jobs</CardTitle>
   431→              <CardDescription>{filtered.length} jobs</CardDescription>
   432→            </div>
   433→            <div className="flex flex-col gap-2 sm:flex-row">
   434→              <div className="relative">
   435→                <Search className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   436→                <Input
   437→                  placeholder="Search jobs…"
   438→                  value={search}
   439→                  onChange={(e) => setSearch(e.target.value)}
   440→                  className="h-9 pl-8 sm:w-56"
   441→                />
   442→              </div>
   443→              <Select
   444→                value={statusFilter}
   445→                onValueChange={(v) => setStatusFilter(v as "all" | JobStatus)}
   446→                disabled={showArchived}
   447→              >
   448→                <SelectTrigger className="h-9 w-full sm:w-40">
   449→                  <Filter className="mr-1.5 h-3.5 w-3.5" />
   450→                  <SelectValue />
   451→                </SelectTrigger>
   452→                <SelectContent>
   453→                  <SelectItem value="all">All statuses</SelectItem>
   454→                  {STATUSES.map((s) => (
   455→                    <SelectItem key={s} value={s}>
   456→                      {s}
   457→                    </SelectItem>
   458→                  ))}
   459→                </SelectContent>
   460→              </Select>
   461→              <Button
   462→                variant={showArchived ? "default" : "outline"}
   463→                size="sm"
   464→                className="h-9 gap-1.5"
   465→                onClick={() => setShowArchived((s) => !s)}
   466→              >
   467→                <Archive className="h-3.5 w-3.5" />
   468→                {showArchived ? "Showing archived" : "Archived"}
   469→              </Button>
   470→            </div>
   471→          </div>
   472→        </CardHeader>
   473→        <CardContent className="p-0">
   474→          {isLoading ? (
   475→            <div className="flex items-center justify-center gap-2 py-16 text-muted-foreground">
   476→              <Loader2 className="h-5 w-5 animate-spin" />
   477→              Loading jobs…
   478→            </div>
   479→          ) : filtered.length === 0 ? (
   480→            <div className="flex flex-col items-center gap-3 py-16 text-center">
   481→              <div className="rounded-full bg-muted p-4">
   482→                <ClipboardList className="h-8 w-8 text-muted-foreground" />
   483→              </div>
   484→              <div>
   485→                <p className="font-medium">No job orders yet</p>
   486→                <p className="text-sm text-muted-foreground">
   487→                  Create a job order to start tracking production.
   488→                </p>
   489→              </div>
   490→            </div>
   491→          ) : (
   492→            <div className="max-h-[65vh] overflow-auto scrollbar-warm">
   493→              <Table>
   494→                <TableHeader className="sticky top-0 z-10 bg-card shadow-sm">
   495→                  <TableRow className="hover:bg-transparent">
   496→                    <TableHead className="w-10">
   497→                      <Checkbox
   498→                        checked={
   499→                          filtered.length > 0 &&
   500→                          filtered.every((j) => selectedIds.has(j.id))
   501→                        }
   502→                        onCheckedChange={(checked) => {
   503→                          if (checked) {
   504→                            setSelectedIds(new Set(filtered.map((j) => j.id)));
   505→                          } else {
   506→                            setSelectedIds(new Set());
   507→                          }
   508→                        }}
   509→                        aria-label="Select all"
   510→                      />
   511→                    </TableHead>
   512→                    <TableHead className="min-w-[140px]">Order</TableHead>
   513→                    <TableHead className="min-w-[110px]">Customer</TableHead>
   514→                    <TableHead className="hidden md:table-cell min-w-[90px]">
   515→                      Assigned
   516→                    </TableHead>
   517→                    <TableHead className="min-w-[80px]">Priority</TableHead>
   518→                    <TableHead className="min-w-[100px]">Status</TableHead>
   519→                    <TableHead className="hidden sm:table-cell min-w-[90px]">
   520→                      Deliver
   521→                    </TableHead>
   522→                    <TableHead className="text-right min-w-[52px]">Manage</TableHead>
   523→                  </TableRow>
   524→                </TableHeader>
   525→                <TableBody>
   526→                  {filtered.map((job, idx) => (
   527→                    <TableRow
   528→                      key={job.id}
   529→                      className={`cursor-pointer hover:bg-muted/50 ${idx % 2 === 1 ? "bg-muted/40" : ""}`}
   530→                      onClick={() => {
   531→                        setSelectedJobId(job.id);
   532→                        setDetailOpen(true);
   533→                      }}
   534→                    >
   535→                      <TableCell onClick={(e) => e.stopPropagation()}>
   536→                        <Checkbox
   537→                          checked={selectedIds.has(job.id)}
   538→                          onCheckedChange={(checked) => {
   539→                            setSelectedIds((prev) => {
   540→                              const next = new Set(prev);
   541→                              if (checked) next.add(job.id);
   542→                              else next.delete(job.id);
   543→                              return next;
   544→                            });
   545→                          }}
   546→                          aria-label={`Select ${job.orderNumber}`}
   547→                        />
   548→                      </TableCell>
   549→                      <TableCell>
   550→                        <div>
   551→                          <p className="font-mono text-xs text-primary">
   552→                            {job.orderNumber}
   553→                          </p>
   554→                          <p className="text-sm font-medium">{job.title}</p>

   745→function CreateJobDialog({
   746→  open,
   747→  onOpenChange,
   748→  customers,
   749→  technicians,
   750→}: {
   751→  open: boolean;

   767→        title: title.trim(),
