    25→// ---- helpers ----
    26→function monthLabel(d: Date) {
    27→  return d.toLocaleDateString("en-US", { month: "long", year: "numeric" });
    28→}
    29→function daysInMonth(d: Date) {
    30→  return new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
    31→}
    32→function pad(n: number) { return String(n).padStart(2, "0"); }
    33→function fmtMonth(d: Date) { return `${d.getFullYear()}-${pad(d.getMonth() + 1)}`; }
    34→function fmtDate(y: number, m: number, day: number) {
    35→  return `${y}-${pad(m + 1)}-${pad(day)}`;
    36→}
    37→function isWeekend(y: number, m: number, day: number) {
    38→  const dow = new Date(y, m, day).getDay();
    39→  return dow === 0 || dow === 6;
    40→}
    41→function statusInfo(status: string) {
    42→  return ATTENDANCE_STATUSES.find((s) => s.value === status) ?? ATTENDANCE_STATUSES[0];
    43→}

    97→function CalendarTab({ canEdit }: { canEdit: boolean }) {
    98→  const qc = useQueryClient();
    99→  const [currentMonth, setCurrentMonth] = React.useState(() => {
   100→    const d = new Date();
   101→    return new Date(d.getFullYear(), d.getMonth(), 1);
   102→  });
   103→  const [selectedJob, setSelectedJob] = React.useState<string>("all");
   104→  const [editingCell, setEditingCell] = React.useState<{
   105→    workerId: string;
   106→    workerName: string;
   107→    date: string;
   108→    record?: AttendanceRecord;
   109→  } | null>(null);
   110→
   111→  const y = currentMonth.getFullYear();
   112→  const m = currentMonth.getMonth();
   113→  const dim = daysInMonth(currentMonth);
   114→  const days = Array.from({ length: dim }, (_, i) => i + 1);
   115→
   116→  const { data: workersData } = useQuery({
   117→    queryKey: ["workers", "active"],
   118→    queryFn: () => workersApi.list({ status: "active" }),
   119→  });
   120→  const workers = workersData?.workers ?? [];
   121→
   122→  const { data: jobsData } = useQuery({
   123→    queryKey: ["jobs", "for-attendance"],
   124→    queryFn: () => jobsApi.list(),
   125→  });
   126→  const jobs = (jobsData as any)?.jobs ?? [];
   127→
   128→  const { data: attData, isLoading } = useQuery({
   129→    queryKey: ["attendance", "month", fmtMonth(currentMonth)],
   130→    queryFn: () => attendanceApi.list({ month: fmtMonth(currentMonth) }),
   131→  });
   132→
   133→  const records = attData?.records ?? [];
   134→  const summary = attData?.summary ?? [];
   135→
   136→  // Build lookup: { workerId_date: record }
   137→  const recordMap = React.useMemo(() => {
   138→    const map: Record<string, AttendanceRecord> = {};
   139→    for (const r of records) {
   140→      const d = new Date(r.date);
   141→      const key = `${r.workerId}_${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
   142→      map[key] = r;

   145→  }, [records]);

   147→  // Summary lookup by workerId
   148→  const summaryMap = React.useMemo(() => {
   149→    const map: Record<string, any> = {};

   152→  }, [summary]);

   154→  const upsertMut = useMutation({
   155→    mutationFn: (payload: Partial<AttendanceRecord> & { workerId: string; date: string }) =>
   156→      attendanceApi.upsert(payload),
   157→    onSuccess: () => {
   158→      qc.invalidateQueries({ queryKey: ["attendance", "month", fmtMonth(currentMonth)] });
   159→    },
   160→    onError: () => toast.error("Failed to save attendance"),
   161→  });
   162→
   163→  function quickSet(workerId: string, workerName: string, day: number, status: string) {
   164→    if (!canEdit) return;
   165→    const date = fmtDate(y, m, day);
   166→    const existing = recordMap[`${workerId}_${date}`];
   167→    upsertMut.mutate({
   168→      workerId,
   169→      date,
   170→      status,
   171→      clockIn: existing?.clockIn ?? null,
   172→      clockOut: existing?.clockOut ?? null,
   173→      jobId: selectedJob !== "all" ? selectedJob : existing?.jobId ?? null,
   174→    });
   175→  }
   176→
   177→  function exportCSV() {
   178→    const header = ["Worker", "Code", "Role", ...days.map((d) => pad(d)), "Total Days", "Total Hours"];
   179→    const rows = workers.map((w) => {
   180→      const s = summaryMap[w.id];
   181→      const cells = days.map((d) => {
   182→        const r = recordMap[`${w.id}_${fmtDate(y, m, d)}`];
   183→        return r ? statusInfo(r.status).symbol : "";
   184→      });
   185→      return [w.name, w.code ?? "", w.role ?? "", ...cells, String(s?.totalDays ?? 0), String(s?.totalHours ?? 0)];
   186→    });
   187→    const csv = [header, ...rows].map((r) => r.map((c) => `"${c}"`).join(",")).join("\n");
   188→    const blob = new Blob([csv], { type: "text/csv" });
   189→    const url = URL.createObjectURL(blob);
   190→    const a = document.createElement("a");
   191→    a.href = url;
   192→    a.download = `attendance-${fmtMonth(currentMonth)}.csv`;
   193→    a.click();
   194→    URL.revokeObjectURL(url);
   195→  }
   196→
   197→  function printReport() {

   199→    if (!w) return;
   200→    const rows = workers.map((wk) => {
   201→      const s = summaryMap[wk.id];
   202→      const cells = days.map((d) => {
   203→        const r = recordMap[`${wk.id}_${fmtDate(y, m, d)}`];
   204→        return `<td style="text-align:center">${r ? statusInfo(r.status).symbol : ""}</td>`;
   205→      }).join("");

   209→    w.document.write(`<html><head><title>Attendance ${monthLabel(currentMonth)}</title>
   210→      <style>body{font-family:Arial;padding:16px;font-size:10px}table{border-collapse:collapse;width:100%}th,td{border:1px solid #ccc;padding:2px 3px}th{background:#f0f0f0}h1{font-size:16px}</style>
   211→      </head><body><h1>Attendance — ${monthLabel(currentMonth)}</h1>
   212→      <table><tr><th>Worker</th><th>Code</th>${dayHeaders}<th>Days</th><th>Hrs</th></tr>${rows}</table>
   213→      <p style="margin-top:12px">Legend: P=Present A=Absent H=Half Day SL=Short Leave FL=Full Leave S=Sick HO=Holiday</p>
   214→      </body></html>`);
   215→    w.document.close();
   216→    w.print();
   217→  }
   218→
   219→  return (
   220→    <div className="space-y-3">
   221→      {/* Toolbar */}
   222→      <div className="flex flex-wrap items-center gap-2 rounded-lg border border-border bg-muted/30 p-2">
   223→        <div className="flex items-center gap-1">
   224→          <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setCurrentMonth(d => new Date(d.getFullYear(), d.getMonth() - 1, 1))}>
   225→            <ChevronLeft className="h-4 w-4" />
   226→          </Button>
   227→          <span className="min-w-[140px] text-center text-sm font-semibold">{monthLabel(currentMonth)}</span>
   228→          <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setCurrentMonth(d => new Date(d.getFullYear(), d.getMonth() + 1, 1))}>
   229→            <ChevronRight className="h-4 w-4" />
   230→          </Button>
   231→          <Button variant="ghost" size="sm" className="h-8 ml-1" onClick={() => { const d = new Date(); setCurrentMonth(new Date(d.getFullYear(), d.getMonth(), 1)); }}>
   232→            Today
   233→          </Button>
   234→        </div>
   235→
   236→        <div className="flex items-center gap-2 ml-auto">
   237→          <Select value={selectedJob} onValueChange={setSelectedJob}>
   238→            <SelectTrigger className="h-8 w-48 text-xs">
   239→              <Briefcase className="h-3 w-3 mr-1" />
   240→              <SelectValue />
   241→            </SelectTrigger>
   242→            <SelectContent>
   243→              <SelectItem value="all">All jobs</SelectItem>
   244→              {jobs.map((j: any) => (
   245→                <SelectItem key={j.id} value={j.id}>{j.orderNumber} — {j.title}</SelectItem>
   246→              ))}
   247→            </SelectContent>
   248→          </Select>
   249→          <Button variant="outline" size="sm" className="h-8" onClick={exportCSV} disabled={workers.length === 0}>
   250→            <Download className="mr-1 h-3.5 w-3.5" /> CSV
   251→          </Button>
   252→          <Button variant="outline" size="sm" className="h-8" onClick={printReport} disabled={workers.length === 0}>
   253→            <Printer className="mr-1 h-3.5 w-3.5" /> Print
   254→          </Button>
   255→        </div>
   256→      </div>
   257→
   258→      {/* Legend */}
   259→      <div className="flex flex-wrap items-center gap-2 text-xs">
   260→        <span className="text-muted-foreground">Legend:</span>
   261→        {ATTENDANCE_STATUSES.map((s) => (
   262→          <span key={s.value} className={`rounded border px-1.5 py-0.5 text-[10px] font-bold ${s.color}`}>
   263→            {s.symbol} = {s.label}
   264→          </span>
   265→        ))}
   266→        <span className="ml-auto text-muted-foreground">
   267→          {workers.length} workers · Click a cell to set status
   268→        </span>
   269→      </div>
   270→
   271→      {/* Calendar grid */}
   272→      {isLoading ? (
   273→        <div className="flex items-center justify-center py-12">
   274→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   275→        </div>
   276→      ) : workers.length === 0 ? (
   277→        <div className="flex flex-col items-center gap-3 py-12 text-center">
   278→          <div className="rounded-full bg-muted p-4">
   279→            <Users className="h-8 w-8 text-muted-foreground" />
   280→          </div>
   281→          <div>
   282→            <p className="font-medium">No workers yet</p>
   283→            <p className="text-sm text-muted-foreground">Add workers in the Workers tab to start tracking attendance.</p>
   284→          </div>
   285→        </div>
   286→      ) : (
   287→        <div className="overflow-x-auto rounded-lg border border-border">
   288→          <table className="w-full border-collapse text-xs">
   289→            <thead>
   290→              <tr className="bg-muted/50">
   291→                <th className="sticky left-0 z-10 border-b border-r border-border bg-muted/50 px-2 py-1.5 text-left font-semibold min-w-[140px]">Worker</th>
   292→                {days.map((d) => (
   293→                  <th key={d} className={`border-b border-r border-border px-0.5 py-1 text-center font-mono text-[10px] ${isWeekend(y, m, d) ? "bg-amber-50 dark:bg-amber-950/20" : ""}`}>
   294→                    {d}
   295→                  </th>
   296→                ))}
   297→                <th className="border-b border-l border-border px-2 py-1 text-center font-semibold">Σ Days</th>
   298→                <th className="border-b border-l border-border px-2 py-1 text-center font-semibold">Σ Hrs</th>
   299→              </tr>
   300→            </thead>
   301→            <tbody>
   302→              {workers.map((w) => {
   303→                const s = summaryMap[w.id];
   304→                return (
   305→                  <tr key={w.id} className="hover:bg-muted/20">
   306→                    <td className="sticky left-0 z-10 border-r border-border bg-card px-2 py-1">
   307→                      <div className="font-medium truncate">{w.name}</div>
   308→                      <div className="text-[9px] text-muted-foreground">{w.code ?? ""} {w.role ? `· ${w.role}` : ""}</div>
   309→                    </td>
   310→                    {days.map((d) => {
   311→                      const date = fmtDate(y, m, d);
   312→                      const r = recordMap[`${w.id}_${date}`];
   313→                      const si = r ? statusInfo(r.status) : null;
   314→                      return (
   315→                        <td
   316→                          key={d}
   317→                          className={`border-r border-border p-0 text-center ${isWeekend(y, m, d) ? "bg-amber-50/50 dark:bg-amber-950/10" : ""}`}
   318→                        >
   319→                          <button
   320→                            disabled={!canEdit}
   321→                            onClick={() => setEditingCell({ workerId: w.id, workerName: w.name, date, record: r })}
   322→                            className={`flex h-7 w-full items-center justify-center text-[10px] font-bold transition-colors ${
   323→                              si ? si.color : "text-transparent hover:bg-muted"
   324→                            } ${canEdit ? "cursor-pointer" : "cursor-default"}`}
   325→                            title={r ? `${si?.label}${r.clockIn ? ` · ${r.clockIn}-${r.clockOut}` : ""}${r.hoursWorked ? ` · ${r.hoursWorked}h` : ""}` : "Click to set"}
   326→                          >
   327→                            {si?.symbol ?? ""}
   328→                          </button>
   329→                        </td>
   330→                      );
   331→                    })}
   332→                    <td className="border-l border-border px-2 py-1 text-center font-bold tabular-nums">{s?.totalDays ?? 0}</td>
   333→                    <td className="border-l border-border px-2 py-1 text-center font-bold tabular-nums">{s?.totalHours ?? 0}</td>
   334→                  </tr>
   335→                );
   336→              })}
   337→            </tbody>
   338→          </table>
   339→        </div>
   340→      )}
   341→
   342→      {/* Quick-set buttons (for selected worker/day) */}
   343→      {canEdit && editingCell && (
   344→        <CellEditDialog
   345→          cell={editingCell}
   346→          jobs={jobs}
   347→          onClose={() => setEditingCell(null)}
   348→          onSave={(payload) => {
   349→            upsertMut.mutate(payload);
   350→            setEditingCell(null);
   351→          }}
   352→          onClear={() => {
   353→            if (editingCell.record) {
   354→              attendanceApi.remove(editingCell.record.id).then(() => {
   355→                qc.invalidateQueries({ queryKey: ["attendance", "month", fmtMonth(currentMonth)] });
   356→                toast.success("Attendance cleared");
   357→              });
   358→            }
   359→            setEditingCell(null);
   360→          }}
   361→        />
   362→      )}
   363→    </div>
   364→  );
   365→}

   367→/* ---- Cell edit dialog — click a cell to open ---- */
   368→function CellEditDialog({
   369→  cell,
   370→  jobs,
   371→  onClose,
   372→  onSave,
   373→  onClear,
   374→}: {
   375→  cell: { workerId: string; workerName: string; date: string; record?: AttendanceRecord };
   376→  jobs: any[];
   377→  onClose: () => void;
   378→  onSave: (payload: Partial<AttendanceRecord> & { workerId: string; date: string }) => void;
   379→  onClear: () => void;
   380→}) {
   381→  const [status, setStatus] = React.useState(cell.record?.status ?? "present");
   382→  const [clockIn, setClockIn] = React.useState(cell.record?.clockIn ?? "08:00");
   383→  const [clockOut, setClockOut] = React.useState(cell.record?.clockOut ?? "17:00");
   384→  const [breakStart, setBreakStart] = React.useState(cell.record?.breakStart ?? "12:00");
   385→  const [breakEnd, setBreakEnd] = React.useState(cell.record?.breakEnd ?? "12:30");
   386→  const [jobId, setJobId] = React.useState(cell.record?.jobId ?? "none");
   387→  const [workLocation, setWorkLocation] = React.useState(cell.record?.workLocation ?? "");
   388→  const [notes, setNotes] = React.useState(cell.record?.notes ?? "");
   389→
   390→  // Auto-calc hours preview
   391→  function calcHours() {
   392→    const toMin = (t: string) => { const m = t.match(/^(\d{1,2}):(\d{2})$/); return m ? parseInt(m[1]) * 60 + parseInt(m[2]) : null; };
   393→    const ci = toMin(clockIn); const co = toMin(clockOut);
   394→    if (ci === null || co === null || co <= ci) return 0;
   395→    let gross = co - ci;
   396→    const bs = toMin(breakStart); const be = toMin(breakEnd);
   397→    if (bs !== null && be !== null && be > bs) gross -= (be - bs);
   398→    return Math.round((gross / 60) * 100) / 100;
   399→  }
   400→
   401→  const hours = calcHours();
   402→
   403→  function handleSave() {
   404→    onSave({
   405→      workerId: cell.workerId,
   406→      date: cell.date,
   407→      status,
   408→      clockIn: clockIn || null,
   409→      clockOut: clockOut || null,
   410→      breakStart: breakStart || null,
   411→      breakEnd: breakEnd || null,
   412→      jobId: jobId && jobId !== "none" ? jobId : null,
   413→      workLocation: workLocation || null,
   414→      notes: notes || null,
   415→    });
   416→  }
   417→
   418→  return (
   419→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   420→      <DialogContent showCloseButton={false} aria-describedby={undefined} className="sm:max-w-[420px]">
   421→        <DialogHeader>
   422→          <DialogTitle className="flex items-center gap-2">
   423→            <Clock className="h-5 w-5 text-primary" />
   424→            {cell.workerName}
   425→          </DialogTitle>
   426→          <DialogDescription>
   427→            {new Date(cell.date).toLocaleDateString("en-US", { weekday: "long", month: "short", day: "numeric" })}
   428→          </DialogDescription>
   429→        </DialogHeader>
   430→
   431→        <div className="space-y-3 py-2">
   432→          {/* Status — click-to-select buttons */}
   433→          <div className="space-y-1.5">
   434→            <Label className="text-xs">Status</Label>
   435→            <div className="flex flex-wrap gap-1">
   436→              {ATTENDANCE_STATUSES.map((s) => (
   437→                <button
   438→                  key={s.value}
   439→                  onClick={() => setStatus(s.value)}
   440→                  className={`rounded border px-2 py-1 text-[11px] font-bold transition-all ${
   441→                    status === s.value ? `${s.color} ring-2 ring-offset-1 ring-primary` : "border-border bg-card text-muted-foreground hover:bg-muted"
   442→                  }`}
   443→                >
   444→                  {s.symbol} {s.label}
   445→                </button>
   446→              ))}
   447→            </div>
   448→          </div>
   449→
   450→          {/* Time fields */}
   451→          <div className="grid grid-cols-2 gap-2">
   452→            <div className="space-y-1">
   453→              <Label className="text-xs">Time In</Label>
   454→              <Input type="time" value={clockIn} onChange={(e) => setClockIn(e.target.value)} className="h-9 text-sm" />
   455→            </div>
   456→            <div className="space-y-1">
   457→              <Label className="text-xs">Time Out</Label>
   458→              <Input type="time" value={clockOut} onChange={(e) => setClockOut(e.target.value)} className="h-9 text-sm" />
   459→            </div>
   460→            <div className="space-y-1">
   461→              <Label className="text-xs">Break Start</Label>
   462→              <Input type="time" value={breakStart} onChange={(e) => setBreakStart(e.target.value)} className="h-9 text-sm" />
   463→            </div>
   464→            <div className="space-y-1">
   465→              <Label className="text-xs">Break End</Label>
   466→              <Input type="time" value={breakEnd} onChange={(e) => setBreakEnd(e.target.value)} className="h-9 text-sm" />
   467→            </div>
   468→          </div>
   469→
   470→          {/* Hours preview */}
