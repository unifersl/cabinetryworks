    25→// ---- helpers ----
    26→function monthLabel(d: Date) {
    27→  return d.toLocaleDateString("en-US", { month: "long", year: "numeric" });
    28→}
    29→function daysInMonth(d: Date) {
    30→  return new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
    31→}
    32→function pad(n: number) { return String(n).padStart(2, "0"); }
    33→function fmtMonth(d: Date) { return `${d.getFullYear()}-${pad(d.getMonth() + 1)}`; }
    34→function fmtDate(y: number, m: number, day: number) {
    35→  return `${y}-${pad(m + 1)}-${pad(day)}`;
    36→}
    37→function isWeekend(y: number, m: number, day: number) {
    38→  const dow = new Date(y, m, day).getDay();
    39→  return dow === 0 || dow === 6;
    40→}
    41→function statusInfo(status: string) {
    42→  return ATTENDANCE_STATUSES.find((s) => s.value === status) ?? ATTENDANCE_STATUSES[0];
    43→}

   350→                          </td>

   352→                      })}
   353→                      <td className="sticky right-0 z-10 border-l border-border px-2 py-1 text-center font-bold tabular-nums text-xs bg-muted/30">
   354→                        {s?.totalDays ?? 0}d / {s?.totalHours ?? 0}h
   355→                      </td>
   356→                    </tr>
   357→                  );
   358→                })}
   359→              </tbody>
   360→            </table>

   362→        </div>

   364→
   365→      {/* Quick-set buttons (for selected worker/day) */}
   366→      {canEdit && editingCell && (

   380→              });
   381→            }
   382→            setEditingCell(null);
   383→          }}
   384→        />
   385→      )}
   386→    </div>
   387→  );

   390→/* ---- Cell edit dialog — click a cell to open ---- */
   391→function CellEditDialog({
   392→  cell,
   393→  jobs,
   394→  onClose,
   395→  onSave,
   396→  onClear,
   397→}: {
   398→  cell: { workerId: string; workerName: string; date: string; record?: AttendanceRecord };
   399→  jobs: any[];
   400→  onClose: () => void;
   401→  onSave: (payload: Partial<AttendanceRecord> & { workerId: string; date: string }) => void;
   402→  onClear: () => void;
   403→}) {
   404→  const [status, setStatus] = React.useState(cell.record?.status ?? "present");
   405→  const [clockIn, setClockIn] = React.useState(cell.record?.clockIn ?? "08:00");
   406→  const [clockOut, setClockOut] = React.useState(cell.record?.clockOut ?? "17:00");
   407→  const [breakStart, setBreakStart] = React.useState(cell.record?.breakStart ?? "12:00");
   408→  const [breakEnd, setBreakEnd] = React.useState(cell.record?.breakEnd ?? "12:30");
   409→  const [jobId, setJobId] = React.useState(cell.record?.jobId ?? "none");
   410→  const [workLocation, setWorkLocation] = React.useState(cell.record?.workLocation ?? "");
   411→  const [notes, setNotes] = React.useState(cell.record?.notes ?? "");
   412→
   413→  // Auto-calc hours preview
   414→  function calcHours() {
   415→    const toMin = (t: string) => { const m = t.match(/^(\d{1,2}):(\d{2})$/); return m ? parseInt(m[1]) * 60 + parseInt(m[2]) : null; };
   416→    const ci = toMin(clockIn); const co = toMin(clockOut);
   417→    if (ci === null || co === null || co <= ci) return 0;
   418→    let gross = co - ci;
   419→    const bs = toMin(breakStart); const be = toMin(breakEnd);
   420→    if (bs !== null && be !== null && be > bs) gross -= (be - bs);
   421→    return Math.round((gross / 60) * 100) / 100;
   422→  }
   423→
   424→  const hours = calcHours();
   425→
   426→  function handleSave() {
   427→    onSave({
   428→      workerId: cell.workerId,
   429→      date: cell.date,
   430→      status,
   431→      clockIn: clockIn || null,
   432→      clockOut: clockOut || null,
   433→      breakStart: breakStart || null,
   434→      breakEnd: breakEnd || null,
   435→      jobId: jobId && jobId !== "none" ? jobId : null,
   436→      workLocation: workLocation || null,
   437→      notes: notes || null,
   438→    });
   439→  }
   440→
   441→  return (
   442→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   443→      <DialogContent showCloseButton={false} aria-describedby={undefined} className="sm:max-w-[420px]">
   444→        <DialogHeader>
   445→          <DialogTitle className="flex items-center gap-2">
   446→            <Clock className="h-5 w-5 text-primary" />
   447→            {cell.workerName}
   448→          </DialogTitle>
   449→          <DialogDescription>
   450→            {new Date(cell.date).toLocaleDateString("en-US", { weekday: "long", month: "short", day: "numeric" })}
   451→          </DialogDescription>
   452→        </DialogHeader>
   453→
   454→        <div className="space-y-3 py-2">
   455→          {/* Status — click-to-select buttons */}
   456→          <div className="space-y-1.5">
   457→            <Label className="text-xs">Status</Label>
   458→            <div className="flex flex-wrap gap-1">
   459→              {ATTENDANCE_STATUSES.map((s) => (
   460→                <button
